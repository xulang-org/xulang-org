Window lifecycle and event loop

#window provides window lifecycle management and per-window input event queues (keyboard, pointer, wheel, and text/IME). Portable event definitions and constants are provided in the pure Xu package window/events.

Call window.pump() once per update loop, then drain each window via target.poll() until EVENT_NONE. Close requests latch in state().close_requested.

@main
#window
#window/events

fn main() {
    options := window.Options{
        title: "Hello Window",
        width: 640,
        height: 360,
        resizable: true,
        high_pixel_density: true,
        hidden: false,
    }
    $target := window.open(&options) or {
        println("Failed to open window")
        return
    }
    println("Window opened: id=", target.id())
    window.pump() or >>
    state := target.state() or >>
    println("Window size:", state.logical_width, "x", state.logical_height)
    target.close() or >>
}

Expected output: prints the window ID and logical dimensions.
