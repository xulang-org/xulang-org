05 · 錯誤處理

不要 Result / Ok / Err / ?。>> expr 丟擲 errors.Error；呼叫處 or (err) { } 接住。

原始碼

@main

#errors

fn read_name(ok bool) string {
    if !ok {
        >> errors.new("empty name")
    }
    << "ada"
}

fn load_or_fallback(ok bool) string {
    << read_name(ok) or (err) {
        println(err.msg())
        << "fallback"
    }
}

fn main() int {
    s := load_or_fallback(false)
    println(s)

    t := read_name(true) or (err) {
        println(err.msg())
        << 1
    }
    println(t)
    << 0
}

編譯（examples 倉庫根，使用已安裝的 xu）：

xu run examples/05_errors/main.xu
# 或：xu build examples/05_errors/main.xu
