03 · 控制流

條件/迴圈無括號。迴圈和 match 的區間統一用 :，含起點、不含終點，即 [start, end)。語句首 -> 是 continue，<- 是 break。

原始碼

@main

fn main() int {
    age := 16
    if age >= 18 {
        println("adult")
    } else {
        println("minor")
    }

    match age {
        0:13 {
            println("child")
        }
        13:20 {
            println("teen")
        }
        else {
            println("adult")
        }
    }

    $n := 3
    while n > 0 {
        println(n)
        n -= 1
    }

    $sum := 0
    for i in 0:5 {
        if i == 2 {
            ->
        }
        if i == 4 {
            <-
        }
        sum += i
    }
    println(sum)
    << 0
}
