14 · TCP echo

listen_on / accept / 每連線 xu handle。

ln := net.listen_on("tcp", "127.0.0.1:19000") >>> (err) { << 1 }
c := ln.accept() >>> (err) { -> }
xu handle(c)

另一終端：printf ping | nc 127.0.0.1 19000。

Windows 上 TCP dial/listen 返回錯誤佔位，不是靜默成功。

原始碼

@main

#net

// TCP echo。
//   printf ping | nc 127.0.0.1 19000
// UDP 見 15udp_echo。

fn handle(c net.Conn) {
    $conn := c
    buf := conn.read(4096) >>> (err) {
        conn.close() >>> (e2) { << }
        <<
    }
    if buf.len == 0 {
        conn.close() >>> (err) { << }
        <<
    }
    conn.write(&buf) >>> (err) { << }
    conn.close() >>> (err) { << }
}

fn main() int {
    ln := net.listen_on("tcp", "127.0.0.1:19000") >>> (err) {
        println(err.msg())
        << 1
    }
    println("tcp://127.0.0.1:19000  echo")
    $running := true
    while running {
        c := ln.accept() >>> (err) {
            ->
        }
        xu handle(c)
    }
    << 0
}
