05 · Errors

Do not use Result / Ok / Err / ?. >> expr throws errors.Error; catch with or (err) { }.

Source

@main

#errors

fn read_name(ok bool) string {
    if !ok {
        >> errors.new("empty name")
    }
    << "ada"
}

fn load_or_fallback(ok bool) string {
    << read_name(ok) or (err) {
        println(err.msg())
        << "fallback"
    }
}

fn main() int {
    s := load_or_fallback(false)
    println(s)

    t := read_name(true) or (err) {
        println(err.msg())
        << 1
    }
    println(t)
    << 0
}
