07 · Arrays and strings

Dynamic arrays with push. String slices s[a:b] copy the half-open range [a, b). Bounds may be omitted: s[a:], s[:b], s[:]. Interpolation: "hi $name".

Source

@main

fn main() int {
    mut nums := [1, 2, 3]
    nums.push(4)
    println(nums.len)
    for n in nums {
        println(n)
    }

    name := "Ada"
    println("hi $name")
    s := "hello"
    println(s.len)
    println(s[1:4])
    << 0
}
