08 · Enum and match

Tagged union. Construct with Color.Rgb(1, 2, 3). match must be exhaustive or have else. Matching an owning payload consumes the subject.

Source

@main

enum Color {
    Red
    Green
    Rgb(int, int, int)
}

fn show(c Color) {
    match c {
        Color.Red {
            println("red")
        }
        Color.Green {
            println("green")
        }
        Color.Rgb(r, g, b) {
            println(r)
            println(g)
            println(b)
        }
    }
}

fn main() int {
    show(Color.Red)
    show(Color.Green)
    show(Color.Rgb(1, 2, 3))
    << 0
}
