fn inspect_screen(s &mut widgets.Screen) int {
    mut session := xacp.open_session()
    permissions := xacp.PERM_UI_READ | xacp.PERM_UI_MUTATE
    if !xacp.set_permissions(mut &session,permissions) {
        xacp.close_session(mut &session)
        >> errors.new("xacp_screen: failed to set least-privilege UI permissions")
    }
    limits := xacp.default_limits()
    nodes := xacp.ui_tree(mut &*s,mut &session,1,false,32,&limits) or (err) {
        xacp.close_session(mut &session)
        >> err
    }
    println("XACP ui.tree: " + str(nodes.len) + " visible nodes")
    for node in nodes {
        println("  #" + str(node.id) + " kind=" + str(node.kind) + " " + node.text)
    }

    mut snapshot := xacp.ui_screenshot(mut &*s,mut &session,2,1.0,&limits) or (err) {
        xacp.close_session(mut &session)
        >> err
    }
    println("XACP ui.screenshot: " + str(snapshot.pixel_width) + "x" + str(snapshot.pixel_height) + " pixels")
    snapshot.dispose() or (err) {
        xacp.close_session(mut &session)
        >> err
    }

    _ := xacp.ui_act(mut &*s,mut &session,2,3,"focus","",false,&limits) or (err) {
        xacp.close_session(mut &session)
        >> err
    }
    xacp.close_session(mut &session)
    << 0
}

fn (m &mut Model) update(screen &mut widgets.Screen,action widgets.Action) int {
    if action.kind == widgets.ACTION_CHANGE {
        if action.id == m.search { query := screen.text_value(m.search) or >>
            screen.set_tree_filter(m.tree,query) or >> }
        if action.id == m.name { screen.set_field_invalid(m.name,false) or >> }
    }
    if action.kind == widgets.ACTION_ITEM_ACTION { _ := screen.notify("Node action · " + str(int(action.value))) or >> }
    if action.kind == widgets.ACTION_COMMIT && action.id == m.extras[5] { screen.set_text(m.message,"Slider committed · " + str(int(action.value))) or >> }
    if action.kind == widgets.ACTION_NOTICE_ACTION { screen.set_text(m.message,"Notification action · " + str(int(action.value))) or >> }
    if action.kind != widgets.ACTION_ACTIVATE { << 0 }
    if action.id == m.controls_page { m.show_controls = !m.show_controls }
    if action.id == m.extras[19] {
        mut notice := widgets.notice(&"Saved a local configuration")
        notice.action = "Undo"
        _ := screen.enqueue_notice(&notice) or >>
    }

    if action.id == m.theme {
        m.theme_index = (m.theme_index + 1) % 3
        current := screen.styles()
        mut styles := widgets.preset_styles(m.theme_index) or >>
        styles.density = current.density
        screen.set_styles(&styles) or >>
        primary_style := styles.button_variant(widgets.BUTTON_PRIMARY) or >>
        screen.set_control_style(m.task,&primary_style) or >>
    }
    if action.id == m.guides {
        m.guided = !m.guided
        mut appearance := widgets.tree_appearance()
        appearance.guides = m.guided
        screen.set_tree_appearance(m.tree,&appearance) or >>
    }
    if action.id == m.density {
        m.compact = !m.compact
        mut styles := screen.styles()
        styles.density = 1.0
        if m.compact { styles.density = 0.85 }
        screen.set_styles(&styles) or >>
    }
    if action.id == m.effects { m.reduced = !m.reduced
        screen.set_reduced_motion(m.reduced) }
    if action.id == m.failure { m.fail_next = !m.fail_next }
    if action.id == m.task && !m.running {
        name := screen.text_value(m.name) or >>
        if name.len == 0 { screen.set_field_invalid(m.name,true) or >>
            screen.focus(m.name) or >>
            << 0 }
        m.task_id = m.task_id + 1
        m.completed = 0
        m.checksum = 0
        m.running = true
        m.next_work = time.instant()
        screen.begin_task(m.task,m.task_id) or >>
    }
    if action.id == m.cancel && m.running {
        _ := screen.complete_task(m.task,m.task_id,widgets.TASK_CANCELLED) or >>
        m.running = false
        screen.set_text(m.message,"Task cancelled. Completed work remains " + str(m.completed) + "/48.") or >>
    }
    << 0
}
fn (m &mut Model) work(screen &mut widgets.Screen,now time.Instant) int {
    if !m.running || now.before(m.next_work) { << 0 }
    // Each progress update follows an actual bounded work batch. The pause
    // represents an application work queue and never changes the percentage.
    for i in 0:5000 { m.checksum = (m.checksum + i * (m.completed + 1)) % 1000003 }
    m.completed = m.completed + 1
    _ := screen.set_task_progress(m.task,m.task_id,f64(m.completed) / 48.0) or >>
    m.next_work = now.add(time.milliseconds(35) or >>) or >>
    if m.fail_next && m.completed == 18 {
        _ := screen.complete_task(m.task,m.task_id,widgets.TASK_ERROR) or >>
        m.running = false
        m.fail_next = false
        screen.set_text(m.message,"Example failure. Retry starts a new task; stale progress is rejected.") or >>
    } else if m.completed == 48 {
        _ := screen.complete_task(m.task,m.task_id,widgets.TASK_SUCCESS) or >>
        m.running = false
        screen.set_text(m.message,"48/48 work batches completed · checksum " + str(m.checksum)) or >>
    }
    << 0
}

fn next_event(events chan ui.Event,deadline time.Instant) ui.Event {
    if deadline.is_zero() { << <-events }
    delay := deadline.since(time.instant())
    if delay.nanoseconds() <= 0 { << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0} }
    timer := delay.timer()
    select {
        event := <-events { timer.stop()
            << event }
        _ := <-timer.c { timer.stop() }
    }
    << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0}
}
fn scene(events chan ui.Event) int {
    ready := demo.ready(events) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    mut screen := widgets.screen(&ready) or >>
    screen.resize(1000.0,820.0) or >>
    screen.set_min_size(900,720) or >>
    mut model := make_model(mut &screen) or (err) { screen.dispose()
        >> err }
    model.layout(mut &screen) or >>
    _ := screen.render() or >>
    while screen.is_open() {
        now := time.instant()
        model.work(mut &screen,now) or >>
        model.layout(mut &screen) or >>
        mut deadline := screen.advance(now) or (err) { >> errors.new("gallery animation: " + err.msg()) }
        for action in screen.take_actions() { model.update(mut &screen,action) or >> }
        _ := screen.render() or (err) { >> errors.new("gallery render: " + err.msg()) }
        if model.running && (deadline.is_zero() || model.next_work.before(deadline)) { deadline = model.next_work }
        event := next_event(events,deadline) or >>
        mut actions := screen.event(&event) or >>
        mut count := 0
        while screen.is_open() && count < 128 {
            pending := ui.poll_chan(events)
            if pending.kind == ui.EV_NONE { <- }
            more := screen.event(&pending) or >>
            for action in more { actions.push(action) }
            count += 1
        }
        for action in actions { model.update(mut &screen,action) or >> }
    }
    screen.dispose()
    << 0
}
