fn tick_loop(emit_ch chan string) {
    mut n := 0
    while true {
        sleep_ns(3000000000)
        n = n + 1
        webview.bus_emit(emit_ch, "tick", str(n))
    }
}

fn watch_shell(shell_ch chan ui.Event) {
    while true {
        ev := <-shell_ch
        if ev.kind == ui.ev_exit || ev.kind == ui.ev_closed {
            println("shell event kind=" + str(ev.kind))
        } else if ev.kind == ui.ev_loaded {
            println("shell event loaded")
        } else if ev.kind == ui.ev_load_failed {
            println("shell event load_failed")
        } else if ev.kind == ui.ev_ready {
            println("shell event ready")
        }
    }
}

    shell_ch := chan ui.Event{cap: 32}

        println(err.msg())
        << 1
    }
    << 0

fn tick_loop(emit_ch chan string) {
    mut n := 0
    while true {
        // Short sleeps so a blocked/mis-scheduled task cannot sit on a 2s nanosleep
        // that starves sibling work if threading is ever wrong. Emit only — no
        // native_eval/title on this loop (those jam the AppKit main thread).
        sleep_ns(200000000)
        n = n + 1
        if n % 10 == 0 {
            webview.bus_emit(emit_ch, "tick", str(n / 10))
        }
    }
}

// Native app.run blocks main; shell events arrive on event_ch from #ui's host pump.
fn watch_shell(shell_ch chan ui.Event, emit_ch chan string) {
    while true {
        ev := <-shell_ch
        if ev.kind == ui.ev_progress {
            ->
        }
        // Result-bearing shell ops → XUN emit (same post protocol as invoke).
        if ev.kind == ui.ev_clipboard {
            webview.bus_emit(emit_ch, "clipboard", ev.menu_id + "")
        } else if ev.kind == ui.ev_file_picked {
            webview.bus_emit(emit_ch, "file_picked", ev.menu_id + "")
        } else if ev.kind == ui.ev_file_saved {
            webview.bus_emit(emit_ch, "file_saved", ev.menu_id + "")
        } else if ev.kind == ui.ev_folder_picked {
            webview.bus_emit(emit_ch, "folder_picked", ev.menu_id + "")
        } else if ev.kind == ui.ev_screenshot {
            p := ev.menu_id + ""
            webview.bus_emit(emit_ch, "screenshot", p + "")
            // reveal_shot / open_shot: follow-up after shell event (post protocol).
            follow := shot_follow_take()
            if follow == "reveal" && p.len > 0 {
                _ := ui.native_reveal_path(p + "")
            } else if follow == "open" && p.len > 0 {
                _ := ui.native_open_path(p + "")
            }
        } else if ev.kind == ui.ev_alert {
            webview.bus_emit(emit_ch, "alert", ev.menu_id + "")
        } else if ev.kind == ui.ev_confirm {
            webview.bus_emit(emit_ch, "confirm", ev.menu_id + "")
        } else if ev.kind == ui.ev_bounds {
            webview.bus_emit(emit_ch, "bounds", ev.menu_id + "")
        } else if ev.kind == ui.ev_state {
            webview.bus_emit(emit_ch, "state", ev.menu_id + "")
        } else if ev.kind == ui.ev_notify {
            webview.bus_emit(emit_ch, "notify", ev.menu_id + "")
        }
        if ev.kind == ui.ev_menu_clicked {
            println("shell event menu_clicked id=" + ev.menu_id)
        } else if ev.kind == ui.ev_tray_menu_clicked {
            println("shell event tray_menu_clicked id=" + ev.menu_id)
        } else if ev.kind == ui.ev_closed {
            println("shell event closed")
        } else if ev.kind == ui.ev_exit {
            println("shell event exit")
        } else if ev.kind == ui.ev_close_requested {
            println("shell event close_requested")
        } else if ev.kind == ui.ev_file_picked {
            println("shell event file_picked " + ev.menu_id)
        } else if ev.kind == ui.ev_clipboard {
            println("shell event clipboard")
        } else if ev.kind == ui.ev_bounds {
            println("shell event bounds " + ev.menu_id)
        } else if ev.kind == ui.ev_state {
            println("shell event state " + ev.menu_id)
        }
    }
}

fn soft_install_tray() int {
    mut tray_items := []ui.MenuItem{}
    tray_items.push(ui.menu_item("show", "Show"))
    tray_items.push(ui.menu_item("hide", "Hide"))
    tray_items.push(ui.menu_item("quit", "Quit"))
    a := ui.install_tray("Xu", ui.menu(tray_items)) or (err) {
        pack_trace("install_tray soft-fail: " + err.msg())
        << 0
    }
    b := ui.install_tray_icon("brand/xu_app_logo.png") or (err) {
        pack_trace("install_tray_icon soft-fail: " + err.msg())
        << 0
    }
    << a + b
}

fn main() int {
    emit_ch := chan string{cap: 64}
    // Large cap: host floods progress/moved/focus during load; a full channel
    // blocks the IPC worker on `shell_ch <-` and delays invoke by ~1s.
    shell_ch := chan ui.Event{cap: 4096}
    mut app := ui.new_app_events(shell_ch)
    pack_trace("main begin")
    app.set_icon(ui.icon_from_file("brand/xu_app_logo.png"))
    // Stage www before the window so host xu://app/ load is not empty.
    webview.stage_www_root(demo_www()) or (err) {
        pack_trace("stage_www_root fail: " + err.msg())
        println(err.msg())
        << 1
    }
    pack_trace("stage_www_root ok")
    xu watch_shell(shell_ch, emit_ch)
    xu ipc_boot(emit_ch, shell_ch)
    xu tick_loop(emit_ch)
    pack_trace("xu tasks spawned")
    sleep_ns(50000000)
    mut win := app.new_window(ui.WindowOpts{title: "Xu WebView IPC", width: 800, height: 560})
    mut menu_items := []ui.MenuItem{}
    menu_items.push(ui.menu_item("about", "About"))
    menu_items.push(ui.menu_item("quit", "Quit"))
    win.set_menu(ui.menu(menu_items))
    soft_install_tray()
    app.new_tray(ui.TrayOpts{tip: "Xu", icon: ui.icon_default(), menu: ui.menu([]ui.MenuItem{})})
    r := mut &win
    mut wv := webview.attach(r)
    pack_trace("webview.attach ok")
    wv.on("ping")
    wv.on("greet")
    wv.load_url(webview.app_scheme_url()) or (err) {
        pack_trace("load_url fail: " + err.msg())
        println(err.msg())
        << 1
    }
    pack_trace("before app.run")
    println("shell events on App.event_ch / xu://app/ (XUN IPC)")
    app.run() or (err) {
        pack_trace("app.run fail: " + err.msg())
        println(err.msg())
        << 1
    }
    pack_trace("window closed")
    println("window closed")
    << 0
}
