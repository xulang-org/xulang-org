#ui/webview

Xu {{xu_version}} API

import: #ui/webview

counts: 41 fns, 15 methods, 2 types, 0 consts, 0 ifaces

Per-window web content. bind_ipc_requests_www dispatches document-scoped InvokeRequest values. __XU__.invoke supports Promise.id / cancel, timeoutMs and AbortSignal. Requests / responses allow 1 MiB, 64 active dispatch leases and an 8 MiB shared budget; cancellation does not interrupt user code. eval_async, find_async and asynchronous cookie APIs complete through ui.ev_request with results bounded to 64 KiB.

Usage and lifetimes

fn route(request webview.InvokeRequest) string {
    if request.name == "slow" { sleep_ns(300000000) }
    if request.name == "wait" { sleep_ns(1500000000) }
    if request.name == "native" {
        target := ui.CanvasEvent{window_id: request.window_id,canvas_id: request.window_id,generation: request.generation}
        options := ui.RequestOptions{timeout_ms: 2000}
        script := "document.title"
        result := ui.webview_request_wait(&target,ui.REQUEST_WEBVIEW_EVAL,&script,&options) or (err) { << "failed: " + err.msg() }
        if result.status != ui.REQUEST_COMPLETED { << "status: " + str(result.status) }
        << result.value + ""
    }
    << request.payload + ""
}
fn ipc(emit_ch chan string,ech chan ui.Event) {
    webview.bind_ipc_requests_www(":mem:",route,emit_ch,ech) or (err) { eprintln(err.msg())
        << }
}
fn events(ech chan ui.Event,emit_ch chan string) {
    while true { event := <-ech
        if event.kind == ui.EV_EXIT { emit_ch <- ""
            << }
        if event.kind == ui.EV_CLOSED { println("Window closed; pending document requests revoked") }
    }
}
fn run() int {
    ech := chan ui.Event{cap: 1024}
    emit_ch := chan string{cap: 64}
    mut app := ui.new_app_events(ech)
    webview.stage_www_root(":mem:") or >>
    xu ipc(emit_ch,ech)
    xu events(ech,emit_ch)
    mut window := app.new_window(ui.WindowOpts{title: "Xu UI / WebView Request Lifecycle",width: 880,height: 640})
    mut browser := webview.attach(mut &window)
    browser.load_url(webview.app_scheme_url()) or >>
    << app.run() or >>
}
fn main() int { << run() or (err) { eprintln(err.msg())
    << 1 } }

// Frameless + PNG mask + WKWebView (macOS).
//   xu build examples/16ui_webview/shaped_webview --target macos-arm64 --desktop -o /tmp/xu_shaped_wv
//   /tmp/xu_shaped_wv
// Drag from the top ~28px strip; Close calls shell.quit.

fn route(name string, payload string) string {
    if name == "ping" {
        << "pong:" + payload
    }
    if name == "greet" {
        << "hello, " + payload
    }
    if name == "win" {
        if payload == "quit" {
            q := ui.native_quit() or (err) { << "err" }
            << "quit"
        }
        if payload == "minimize" {
            m := ui.native_minimize() or (err) { << "err" }
            << "minimized"
        }
        << "win?"
    }
    << "miss"
}

fn demo_www() string {
    << "www"
}

fn ipc_boot(emit_ch chan string, shell_ch chan ui.Event) {
    webview.bind_ipc_www(demo_www(), route, emit_ch, shell_ch) or (err) {
        eprintln(err.msg())
        <<
    }
}

fn main() int {
    emit_ch := chan string{cap: 32}
    shell_ch := chan ui.Event{cap: 32}
    mut app := ui.new_app_events(shell_ch)
    app.set_icon(ui.icon_from_file("brand/xu_app_logo.png"))
    webview.stage_www_root(demo_www()) or (err) {
        println(err.msg())
        << 1
    }
    xu ipc_boot(emit_ch, shell_ch)
    sleep_ns(50000000)

    mut win := app.new_window(ui.WindowOpts{title: "Xu Shaped WebView", width: 360, height: 360})
    r := mut &win
    mut wv := webview.attach(r)
    wv.on("ping")
    wv.on("greet")
    wv.on("win")

    a0 := ui.native_set_frameless(1)
    a1 := ui.native_set_transparent(1)
    a2 := ui.native_set_window_mask("brand/xu_app_logo.png") or (err) {
        println(err.msg())
        << 1
    }
    a3 := ui.native_set_shadow(1)
    if a0 + a1 + a2 + a3 < 0 {
        << 1
    }

    wv.load_url("xu://app/shaped.html") or (err) {
        println(err.msg())
        << 1
    }
    println("shaped webview — xu://app/shaped.html XUN IPC (drag top; Close quits)")
    app.run() or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

// Remote HTTP document + xu:// IPC.
// 1) python3 examples/16ui_webview/remote_http/serve.py
// 2) xu build examples/16ui_webview/remote --target macos-arm64 --desktop -o /tmp/xu_remote
// 3) /tmp/xu_remote
// Optional: XU_UI_REMOTE_URL=http://127.0.0.1:8765/

fn route(name string, payload string) string {
    if name == "ping" {
        << "pong:" + payload
    }
    if name == "greet" {
        << "hello, " + payload
    }
    << "unknown:" + name
}

fn demo_www() string {
    << "www"
}

fn remote_url() string {
    u := env_get("XU_UI_REMOTE_URL")
    if u.len > 0 {
        << u + ""
    }
    << "http://127.0.0.1:8765/"
}

fn ipc_boot(emit_ch chan string, shell_ch chan ui.Event) {
    // Still stage www + bridge so xu://app/__xu works; main document is remote.
    webview.bind_ipc_www(demo_www(), route, emit_ch, shell_ch) or (err) {
        eprintln(err.msg())
        <<
    }
}

fn tick_loop(emit_ch chan string) {
    mut n := 0
    while true {
        sleep_ns(3000000000)
        n = n + 1
        webview.bus_emit(emit_ch, "tick", str(n))
    }
}

fn watch_shell(shell_ch chan ui.Event) {
    while true {
        ev := <-shell_ch
        if ev.kind == ui.EV_EXIT || ev.kind == ui.EV_CLOSED {
            println("shell event kind=" + str(ev.kind))
        } else if ev.kind == ui.EV_LOADED {
            println("shell event loaded")
        } else if ev.kind == ui.EV_LOAD_FAILED {
            println("shell event load_failed")
        } else if ev.kind == ui.EV_READY {
            println("shell event ready")
        }
    }
}

fn main() int {
    emit_ch := chan string{cap: 32}
    shell_ch := chan ui.Event{cap: 32}
    mut app := ui.new_app_events(shell_ch)
    app.set_icon(ui.icon_from_file("brand/xu_app_logo.png"))
    url := remote_url()
    xu watch_shell(shell_ch)
    xu ipc_boot(emit_ch, shell_ch)
    xu tick_loop(emit_ch)
    sleep_ns(50000000)
    mut win := app.new_window(ui.WindowOpts{title: "Xu Remote HTTP + xu:// IPC", width: 720, height: 520})
    r := mut &win
    mut wv := webview.attach(r)
    wv.on("ping")
    wv.on("greet")
    url2 := url + ""
    wv.load_remote(url2) or (err) {
        println(err.msg())
        << 1
    }
    println("remote document " + url + " — IPC via xu://app/__xu (start remote_http/serve.py first)")
    app.run() or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

Clear cookies asynchronously through ui.ev_request, with cancellation, deadlines and window / navigation ownership. Results are at most 64 KiB.

Parameters

wv &WebView
receiver &WebView (borrow)
options &ui.RequestOptions
options (borrowed; caller keeps it; type &ui.RequestOptions)

Returns

an admitted ui.Request; throws on failure
