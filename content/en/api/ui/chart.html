fn next_event(events chan ui.Event, deadline time.Instant) ui.Event{
    if deadline.is_zero() { << <- events}
    delay := deadline.since(time.instant())
    if delay.nanoseconds() <= 0 { << ui.Event{kind: ui.EV_NONE, window_id: 0, menu_id: "", width: 0, height: 0}}
    timer := delay.timer()
    select {event := <- events{timer.stop()
            << event}
        _ := <- timer.c{timer.stop()}}
    << ui.Event{kind: ui.EV_NONE, window_id: 0, menu_id: "", width: 0, height: 0}
}

fn workshop_loop(events chan ui.Event, model $&Workshop, screen $&widgets.Screen) int {
    model.apply_locale(screen) or >>
    screen.set_text(model.title, "XUUI  /  Field") or >>
    model.configure_controls(screen) or >>
    model.controls_dirty = false
    model.apply_appearance(screen) or >>
    model.appearance_dirty = false
    $dispatcher := register_events(& * model, & * screen) or >>
    model.layout(screen) or >>
    while screen.is_open() {
        register_notice_events(& * screen, $&dispatcher) or >>
        now := time.instant()
        model.work(screen, now) or >>
        if model.controls_dirty{model.configure_controls(screen) or >>
            model.controls_dirty = false}
        if model.appearance_dirty{model.apply_appearance(screen) or >>
            model.appearance_dirty = false}
        model.apply_page_options(screen) or >>
        model.layout(screen) or >>
        $deadline := screen.advance(now) or >>
        $timed_actions := false
        for action in screen.take_actions() {
            _ := dispatcher.dispatch_or_fallback(&action, model, screen) or >>
            timed_actions = true
        }
        // Reconcile timer-driven values/styles before computing the next wait.
        if timed_actions{ -> }
        _ := screen.render() or >>
        if model.running &&(deadline.is_zero() || model.deadline.before(deadline)) {deadline = model.deadline}
        event := next_event(events, deadline) or >>
        $actions := screen.event(&event) or >>
        $count := 0
        while screen.is_open() && count < 128 {pending := ui.poll_chan(events)
            if pending.kind == ui.EV_NONE{ <- }
            more := screen.event(&pending) or >>
            for action in more{actions.push(action)}
            count += 1}
        for action in actions{register_notice_events(& * screen, $&dispatcher) or >>
            _ := dispatcher.dispatch_or_fallback(&action, model, screen) or >> }
    }
    << 0
}

fn chart_data_locale(kind int, revision int, english bool)[]chart.ChartSeries{
    $result :=[]chart.ChartSeries{}
    $series_count := 2
    single := kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
        kind == widgets.CHART_GAUGE || kind == widgets.CHART_HISTOGRAM || kind == widgets.CHART_BOXPLOT ||
        kind == widgets.CHART_CANDLESTICK || kind == widgets.CHART_WATERFALL || kind == widgets.CHART_FUNNEL ||
        (kind >= widgets.CHART_POLAR_BAR && kind <= widgets.CHART_SANKEY) || kind == widgets.CHART_VIOLIN ||
        kind == widgets.CHART_BULLET || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_PARETO
    if single{series_count = 1}
    if kind == widgets.CHART_HEATMAP{series_count = 3}
    for si in 0:series_count{
        $name := "构建"
        if si == 1 {name = "索引"}
        if si == 2 {name = "绘制"}
        if single{name = "工作量"}
        if english {
            name = "Build"
            if si == 1 {name = "Index"}
            if si == 2 {name = "Render"}
            if single{name = "Workload"}
        }
        $series := chart.chart_series(si + 1, name, kind)
        series.smooth = kind == widgets.CHART_LINE || kind == widgets.CHART_AREA || kind == widgets.CHART_STACKED_AREA ||
            kind == widgets.CHART_SMOOTH_LINE || kind == widgets.CHART_SMOOTH_AREA
        $count := 12 - revision % 2
        if kind == widgets.CHART_SCATTER{count = 64}
        else if kind == widgets.CHART_BUBBLE{count = 16}
        else if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{count = 5}
        else if kind == widgets.CHART_RADAR || kind == widgets.CHART_PARETO{count = 6}
        else if kind == widgets.CHART_GAUGE{count = 3}
        else if kind == widgets.CHART_POLAR_BAR{count = 4}
        else if kind == widgets.CHART_HORIZONTAL_BAR || kind == widgets.CHART_HEATMAP || kind == widgets.CHART_BULLET ||
                kind == widgets.CHART_LOLLIPOP || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_VIOLIN{count = 8}
        else if kind == widgets.CHART_WATERFALL{count = 7}
        for i in 0:count{
            x := f64(i + 1)
            phase := f64(revision) * 0.9 + f64(si) * 1.7
            $y := 48.0 + f64(si) * 20.0 + math.sin(x * 0.55 + phase) * 20.0 + math.cos(x * 0.17 + phase) * 7.0
            if kind == widgets.CHART_FUNNEL || kind == widgets.CHART_PARETO || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                y = f64((count - i) * 22 + (revision % 3) * 4)
            } else if kind == widgets.CHART_GAUGE || kind == widgets.CHART_POLAR_BAR{
                y = math.max(12.0, math.min(96.0, 34.0 + f64(i) * 18.0 + math.sin(phase + x) * 14.0))
            } else if kind == widgets.CHART_WATERFALL{
                deltas :=[42.0, 18.0, 0.0 - 14.0, 24.0, 0.0 - 11.0, 16.0, 0.0]
                y = deltas[i] + math.sin(phase) * 3.0
                if i == count - 1 {y = 0.0}
            }
            $point := chart.chart_point(i + 1, x, y)
            if kind == widgets.CHART_BUBBLE{
                point = chart.chart_bubble_point(i + 1, x, y, 12.0 + math.abs(math.cos(x * 0.7 + phase)) * 36.0)
            } else if kind == widgets.CHART_BOXPLOT || kind == widgets.CHART_CANDLESTICK ||
                      kind == widgets.CHART_VIOLIN || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_BULLET{
                $close := y + math.sin(x * 0.9 + phase) * 12.0
                if (kind == widgets.CHART_BOXPLOT || kind == widgets.CHART_VIOLIN || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_BULLET) && close < y{
                    close = y + 10.0
                }
                low := math.min(y, close) - 8.0 - math.abs(math.cos(x + phase)) * 6.0
                high := math.max(y, close) + 8.0 + math.abs(math.sin(x + phase)) * 6.0
                point = chart.chart_range_point(i + 1, x, y, low, high, close)
            } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                point.size = 1.0
            }
            if english {
                point.label = "Sample " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                    point.label =["Compile", "Index", "Render", "I/O", "Other"][i]
                } else if kind == widgets.CHART_RADAR{
                    point.label =["Throughput", "Latency", "Memory", "Startup", "Cache", "Binary"][i]
                } else if kind == widgets.CHART_GAUGE{
                    point.label =["CPU", "Memory", "Cache"][i]
                } else if kind == widgets.CHART_POLAR_BAR{
                    point.label =["Core 0", "Core 1", "Core 2", "Core 3"][i]
                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "Total"
                }
            } else {
                point.label = "样本 " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                    point.label =["编译", "索引", "绘制", "输入输出", "其他"][i]
                } else if kind == widgets.CHART_RADAR{
                    point.label =["吞吐量", "延迟", "内存", "启动", "缓存", "体积"][i]
                } else if kind == widgets.CHART_GAUGE{
                    point.label =["处理器", "内存", "缓存"][i]
                } else if kind == widgets.CHART_POLAR_BAR{
                    point.label =["核心 0", "核心 1", "核心 2", "核心 3"][i]
                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "总计"
                }
            }
            series.points.push(point)
        }
        result.push(series)
    }
    << result
}
