#ui/layout

Xu {{xu_version}} API

import: #ui/layout

counts: 11 fns, 29 methods, 10 types, 19 consts, 0 ifaces

Constraints, sizes, rectangles, row / column / stack / padding and retained layout trees. Constraints flow downward and measured sizes upward. BoxStyle width / height=-1 means natural size. Trees allow 4096 nodes and depth 64; removed node IDs never repeat. View in widgets handles painting and input.

Usage and lifetimes

fn caption(id int) string {
    if id == 0 { << "1 / Copy greeting" }
    if id == 1 { << "2 / Read clipboard" }
    if id == 2 { << "3 / Open files" }
    if id == 3 { << "4 / Choose save path" }
    if id == 4 { << "5 / Choose folders" }
    if id == 5 { << "6 / Prompt (15s timeout)" }
    << "Cancel pending request"
}
fn status_name(status int) string {
    if status == ui.REQUEST_COMPLETED { << "completed" }
    if status == ui.REQUEST_CANCELLED { << "cancelled" }
    if status == ui.REQUEST_TIMED_OUT { << "timed out" }
    if status == ui.REQUEST_UNSUPPORTED { << "unsupported" }
    << "failed"
}
fn submit(identity &ui.CanvasEvent,choice int) ui.Request {
    options := ui.RequestOptions{timeout_ms: 5000}
    if choice == 0 { value := "Hello from Xu UI 0.10 / 你好"
        << ui.clipboard_write_async(identity,&value,&options) or >> }
    if choice == 1 { << ui.clipboard_read_async(identity,&options) or >> }
    if choice == 5 { prompt := ui.PromptOptions{title: "Your greeting",seed: "Hello, Xu",timeout_ms: 15000}
        << ui.prompt_async(identity,&prompt) or >> }
    $files := ui.file_dialog_options()
    if choice == 2 {
        files.title = "Select one or more text files"
        files.multiple = true
        files.filters = [ui.FileFilter{label: "Text",extensions: ["txt","md","xu"]}]
        << ui.open_files_async(identity,&files) or >>
    }
    if choice == 3 { files.title = "Choose a destination (this demo does not write)"
        files.name = "xu-note.txt"
        << ui.save_file_async(identity,&files) or >> }
    files.title = "Choose folders"
    files.multiple = true
    << ui.select_folders_async(identity,&files) or >>
}
fn paint(context &draw.Context,frame $&draw.Frame,message string,pending int) int {
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,28.0,26.0,14.0,demo.accent(),"05 / ASYNCHRONOUS DESKTOP SERVICES") or >>
    demo.text(context,frame,28.0,58.0,28.0,demo.ink(),"A request is not a result.") or >>
    for id in 0:7 {
        y := 120.0 + f64(id) * 48.0
        frame.fill_rect(28.0,y,370.0,38.0,draw.rgb(40,57,73)) or >>
        demo.text(context,frame,40.0,y + 8.0,15.0,demo.ink(),caption(id)) or >>
    }
    demo.text(context,frame,430.0,124.0,17.0,demo.accent(),"Pending request: " + str(pending)) or >>
    $options := draw.text_options(16.0)
    options.width = 360.0
    $label := context.layout_text(&message,&options) or >>
    frame.text_layout(&label,430.0,172.0,demo.ink()) or >>
    demo.text(context,frame,28.0,490.0,13.0,demo.muted(),"One event channel / request IDs / distinct terminal states / Escape to close") or >>
    frame.present() or >>
    label.dispose() or >>
    << 0
}
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    $frame := context.new_frame(4096)
    $pending := ui.Request{id: 0,window_id: 0,canvas_id: 0,generation: 0,kind: 0}
    $message := "Choose an action. File dialogs return paths; selecting a save path does not write a file."
    $armed := -1
    paint(&context,$&frame,message + "",pending.id) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        $dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind == ui.EV_REQUEST {
            result := ui.request_of(&event) or >>
            if result.matches(&pending) {
                message = "Request " + str(pending.id) + " " + status_name(result.status)
                if result.status == ui.REQUEST_COMPLETED {
                    if result.request.kind >= ui.REQUEST_OPEN_FILES {
                        paths := ui.file_paths(&result) or >>
                        for path in paths { message = message + "\n" + path }
                    } else { message = message + "\n" + result.value }
                }
                println("Request " + str(pending.id) + ": " + status_name(result.status))
                pending.id = 0
                dirty = true
            }
        }
        if event.kind == ui.EV_BLURRED { armed = -1 }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            $choice := -1
            for id in 0:7 {
                bounds := layout.Rect{x: 28.0,y: 120.0 + f64(id) * 48.0,width: 370.0,height: 38.0}
                if bounds.contains(input.x,input.y) { choice = id }
            }
            if input.kind == ui.EV_POINTER_DOWN && input.button == ui.BUTTON_LEFT { armed = choice }
            if input.kind == ui.EV_POINTER_CANCEL { armed = -1 }
            $activate := -1
            if input.kind == ui.EV_POINTER_UP && input.button == ui.BUTTON_LEFT {
                if choice == armed { activate = choice }
                armed = -1
            }
            if input.kind == ui.EV_KEY_DOWN && input.repeat == 0 {
                if input.key >= 49 && input.key <= 54 { activate = input.key - 49 }
                if input.key == ui.KEY_ESCAPE { demo.quit() or >> }
            }
            if activate == 6 && pending.id > 0 { _ := pending.cancel()
                message = "Cancellation requested; waiting for terminal result."
                dirty = true }
            if activate >= 0 && activate < 6 {
                if pending.id == 0 {
                    request := submit(&identity,activate) or (err) { message = err.msg()
                        paint(&context,$&frame,message + "",pending.id) or >>
                        -> }
                    pending = request
                    message = "Request admitted. Keep interacting; the event loop remains active."
                } else { message = "Finish or cancel the current request first." }
                dirty = true
            }
        }
        if dirty { paint(&context,$&frame,message + "",pending.id) or >> }
    }
    if pending.id > 0 { _ := pending.cancel() }
    frame.dispose()
    << 0
}
fn worker(ech chan ui.Event) { scene(ech) or (err) { demo.fail(err.msg())
    << } }
fn launch() int {
    $app := demo.app("Xu UI / Desktop Services",850,550) or >>
    xu worker(app.event_ch)
    << app.run() or >>
}

    << 1 } }

fn paint(context &draw.Context,frame $&draw.Frame,input &ui.TextInput) int {
    state := input.snapshot() or >>
    value := state.display_value()
    println("Text revision: " + str(state.revision) + " / display bytes: " + str(value.len))
    options := draw.text_options(28.0)
    $text := context.layout_text(&value,&options) or >>
    text.bind_input(input,state.revision,44.0,180.0) or >>
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,28.0,28.0,14.0,demo.accent(),"04 / NATIVE TEXT INPUT") or >>
    demo.text(context,frame,28.0,60.0,28.0,demo.ink(),"Type in your own language.") or >>
    demo.text(context,frame,28.0,112.0,15.0,demo.muted(),"Use an IME, arrows, Shift+arrows and Backspace. Click to move the caret.") or >>
    frame.fill_rect(28.0,160.0,790.0,86.0,draw.rgb(26,37,50)) or >>
    anchor := text.caret(state.display_anchor(),false) or >>
    caret := text.caret(state.display_focus(),false) or >>
    if state.display_anchor() != state.display_focus() && anchor.y == caret.y {
        frame.fill_rect(44.0 + math.min(anchor.x,caret.x),180.0 + caret.y,math.abs(caret.x - anchor.x),caret.height,draw.rgba(78,205,196,80)) or >>
    }
    frame.text_layout(&text,44.0,180.0,demo.ink()) or >>
    frame.fill_rect(44.0 + caret.x,180.0 + caret.y,1.5,caret.height,demo.accent()) or >>
    if state.composing {
        demo.text(context,frame,28.0,280.0,17.0,demo.accent(),"Composing: " + state.preedit) or >>
    } else {
        demo.text(context,frame,28.0,280.0,17.0,demo.muted(),"Committed text / no active preedit") or >>
    }
    demo.text(context,frame,28.0,324.0,14.0,demo.muted(),"Revision " + str(state.revision) + " / anchor byte " + str(state.anchor) + " / focus byte " + str(state.focus)) or >>
    demo.text(context,frame,28.0,360.0,14.0,demo.muted(),"Selection uses UTF-8 byte offsets. Escape ends the session and closes the window.") or >>
    frame.present() or >>
    text.dispose() or >>
    << 0
}
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    seed := "你好，Xu 👩‍💻 · café"
    caret := ui.TextInputRect{x: 44.0,y: 180.0,width: 2.0,height: 36.0}
    options := ui.text_input_options(&seed,caret)
    $input := ui.start_text_input(identity.canvas_id,identity.generation,&options) or >>
    $frame := context.new_frame(4096)
    paint(&context,$&frame,&input) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        $dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind >= ui.EV_TEXT_COMMIT && event.kind <= ui.EV_TEXT_SURROUNDING_DELETE {
            edit := ui.text_edit_of(&event) or >>
            if edit.input.id == input.id {
                if event.kind == ui.EV_TEXT_INPUT_ENDED || event.kind == ui.EV_TEXT_INPUT_FAILED { <- }
                dirty = true
            }
        }
        if ui.is_input_event(event.kind) {
            pointer := ui.input_of(&event) or >>
            if pointer.generation != identity.generation { -> }
            if pointer.kind == ui.EV_KEY_DOWN && pointer.key == ui.KEY_ESCAPE { demo.quit() or >> }
            bounds := layout.Rect{x: 28.0,y: 160.0,width: 790.0,height: 86.0}
            if pointer.kind == ui.EV_POINTER_DOWN && pointer.button == ui.BUTTON_LEFT && bounds.contains(pointer.x,pointer.y) {
                state := input.snapshot() or >>
                // The native editor owns preedit; do not overwrite an active composition.
                if !state.composing {
                    hit := input.hit_test(state.revision,pointer.x,pointer.y) or >>
                    index := hit.byte_index()
                    _ := input.update(state.revision,&state.value,index,index,&caret) or >>
                    dirty = true
                }
            }
        }
        if dirty { paint(&context,$&frame,&input) or >> }
    }
    frame.dispose()
    input.stop() or (err) { << 0 }
    << 0
}

// One worker owns the controls and their canvas resources. Every recorded
// label remains alive until the complete frame has been accepted.
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    $switcher := widgets.style_switcher()
    switcher.use_preset(widgets.THEME_LIGHT) or >>
    $toggle := widgets.switch_control(&identity,1,&rect(48.0,116.0,280.0,48.0),true) or >>
    $radio := widgets.radio(&identity,2,&rect(48.0,180.0,280.0,48.0),1,false) or >>
    $slider := widgets.slider(&identity,3,&rect(48.0,260.0,280.0,36.0),0.0,100.0,50.0) or >>
    slider.set_step(5.0) or >>
    $stepper := widgets.stepper(&identity,4,&rect(48.0,324.0,280.0,48.0),0.0,20.0,1.0,5.0) or >>
    $tabs := widgets.tabs(&identity,5,&rect(376.0,116.0,348.0,48.0),3,0) or >>
    $dropdown := widgets.select_box(&identity,6,&rect(376.0,196.0,348.0,48.0),3,0) or >>
    $list := widgets.list_box(&identity,7,&rect(376.0,284.0,348.0,132.0),3,0) or >>
    panel := widgets.card(&rect(24.0,92.0,724.0,352.0)) or >>
    badge := widgets.badge(&rect(48.0,392.0,280.0,32.0)) or >>
    divider := widgets.divider(&rect(48.0,244.0,280.0,2.0),widgets.AXIS_HORIZONTAL) or >>
    $label0 := widgets.styled_label(&context,&"Enabled")
    $label1 := widgets.styled_label(&context,&"Choose this option")
    $label2 := widgets.styled_label(&context,&"Stepper")
    $label3 := widgets.styled_label(&context,&"One")
    $label4 := widgets.styled_label(&context,&"Two")
    $label5 := widgets.styled_label(&context,&"Three")
    $label6 := widgets.styled_label(&context,&"Select an option")
    $label7 := widgets.styled_label(&context,&"Xu 0.10.2")
    $label8 := widgets.styled_label(&context,&"One")
    $label9 := widgets.styled_label(&context,&"Two")
    $label10 := widgets.styled_label(&context,&"Three")
    $frame := context.new_frame(16384)
    $dirty := true
    $focus := 0
    while true {
        if dirty {
            styles := switcher.styles()
            frame.begin() or >>
            frame.set_bg(styles.background) or >>
            panel.paint_styled($&frame,&styles) or >>
            ink := styles.button.palette.foreground
            title := "Extended controls / " + switcher.name()
            frame.text(32.0,28.0,22.0,ink,&title) or >>
            toggle.paint_styled($&frame,$&label0,&styles) or >>
            radio.paint_styled($&frame,$&label1,&styles) or >>
            divider.paint_styled($&frame,&styles) or >>
            slider.paint_styled($&frame,&styles) or >>
            stepper.paint_styled($&frame,$&label2,&styles) or >>
            tabs.paint_box_styled($&frame,&styles) or >>
            list.paint_box_styled($&frame,&styles) or >>
            tabs.paint_tab_styled($&frame,0,$&label3,&styles) or >>
            list.paint_item_styled($&frame,0,$&label8,&styles) or >>
            tabs.paint_tab_styled($&frame,1,$&label4,&styles) or >>
            list.paint_item_styled($&frame,1,$&label9,&styles) or >>
            tabs.paint_tab_styled($&frame,2,$&label5,&styles) or >>
            list.paint_item_styled($&frame,2,$&label10,&styles) or >>
            dropdown.paint_styled($&frame,$&label6,&styles) or >>
            badge.paint_styled($&frame,$&label7,&styles) or >>
            frame.text(32.0,464.0,15.0,ink,&"T: theme / S: text size / D: density / Tab: focus / Esc: quit") or >>
            status := "Slider " + str(int(slider.value())) + " / Stepper " + str(int(stepper.value())) + " / Select " + str(dropdown.selected() + 1)
            frame.text(32.0,494.0,15.0,ink,&status) or >>
            // A Select popup overlays the scene; this basic painter repeats
            // its label. The selected option index is shown in the status.
            dropdown.paint_popup_styled($&frame,$&label6,&styles) or >>
            frame.present() or >>
            toggle.painted()
            radio.painted()
            slider.painted()
            stepper.painted()
            tabs.painted()
            dropdown.painted()
            list.painted()
            dirty = false
        }
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT { <- }
        if event.window_id != identity.window_id { -> }
        if event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.kind == ui.EV_CANVAS_METRICS { dirty = true }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            if input.kind == ui.EV_KEY_DOWN && input.repeat == 0 {
                if input.key == ui.KEY_ESCAPE {
                    // App shutdown revokes canvas resources. Do not enqueue
                    // another frame while the native host is closing.
                    demo.quit() or >>
                    << 0
                }
                if input.key == 84 || input.key == 116 { switcher.next() or >>
                    dirty = true }
                if input.key == 83 || input.key == 115 {
                    $scale := 1.4
                    if switcher.text_scale() > 1.0 { scale = 1.0 }
                    switcher.set_text_scale(scale) or >>
                    dirty = true
                }
                if input.key == 68 || input.key == 100 {
                    $density := 0.85
                    if switcher.density() < 1.0 { density = 1.0 }
                    switcher.set_density(density) or >>
                    dirty = true
                }
                if input.key == ui.KEY_TAB { focus = focus % 7 + 1
                    dirty = true }
            }
        }
        // Route the popup first. A modal option click must not also focus
        // or activate the ListBox underneath it.
        popup_open := dropdown.is_open()
        first := dropdown.event(&event) or >>
        $actions := [first]
        if !popup_open && !first.consumed {
            actions.push(toggle.event(&event) or >>)
            actions.push(radio.event(&event) or >>)
            actions.push(slider.event(&event) or >>)
            actions.push(stepper.event(&event) or >>)
            actions.push(tabs.event(&event) or >>)
            actions.push(list.event(&event) or >>)
        }
        for action in actions {
            if action.request_focus { focus = action.id }
            if action.needs_paint { dirty = true }
        }
        toggle.set_focus(focus == 1)
        radio.set_focus(focus == 2)
        slider.set_focus(focus == 3)
        stepper.set_focus(focus == 4)
        tabs.set_focus(focus == 5)
        dropdown.set_focus(focus == 6)
        list.set_focus(focus == 7)
    }
    frame.dispose()
    if demo.alive(&identity) {
        label0.dispose()
        label1.dispose()
        label2.dispose()
        label3.dispose()
        label4.dispose()
        label5.dispose()
        label6.dispose()
        label7.dispose()
        label8.dispose()
        label9.dispose()
        label10.dispose()
    }
    << 0
}
