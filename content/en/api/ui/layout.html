fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    $frame := context.new_frame(4096)
    $pending := ui.Request{id: 0,window_id: 0,canvas_id: 0,generation: 0,kind: 0}
    $message := "Choose an action. File dialogs return paths; selecting a save path does not write a file."
    $armed := -1
    paint(&context,$&frame,message + "",pending.id) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        $dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind == ui.EV_REQUEST {
            result := ui.request_of(&event) or >>
            if result.matches(&pending) {
                message = "Request " + str(pending.id) + " " + status_name(result.status)
                if result.status == ui.REQUEST_COMPLETED {
                    if result.request.kind >= ui.REQUEST_OPEN_FILES {
                        paths := ui.file_paths(&result) or >>
                        for path in paths { message = message + "\n" + path }
                    } else { message = message + "\n" + result.value }
                }
                println("Request " + str(pending.id) + ": " + status_name(result.status))
                pending.id = 0
                dirty = true
            }
        }
        if event.kind == ui.EV_BLURRED { armed = -1 }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            $choice := -1
            for id in 0:7 {
                bounds := layout.Rect{x: 28.0,y: 120.0 + f64(id) * 48.0,width: 370.0,height: 38.0}
                if bounds.contains(input.x,input.y) { choice = id }
            }
            if input.kind == ui.EV_POINTER_DOWN && input.button == ui.BUTTON_LEFT { armed = choice }
            if input.kind == ui.EV_POINTER_CANCEL { armed = -1 }
            $activate := -1
            if input.kind == ui.EV_POINTER_UP && input.button == ui.BUTTON_LEFT {
                if choice == armed { activate = choice }
                armed = -1
            }
            if input.kind == ui.EV_KEY_DOWN && input.repeat == 0 {
                if input.key >= 49 && input.key <= 54 { activate = input.key - 49 }
                if input.key == ui.KEY_ESCAPE { demo.quit() or >> }
            }
            if activate == 6 && pending.id > 0 { _ := pending.cancel()
                message = "Cancellation requested; waiting for terminal result."
                dirty = true }
            if activate >= 0 && activate < 6 {
                if pending.id == 0 {
                    request := submit(&identity,activate) or (err) { message = err.msg()
                        paint(&context,$&frame,message + "",pending.id) or >>
                        -> }
                    pending = request
                    message = "Request admitted. Keep interacting; the event loop remains active."
                } else { message = "Finish or cancel the current request first." }
                dirty = true
            }
        }
        if dirty { paint(&context,$&frame,message + "",pending.id) or >> }
    }
    if pending.id > 0 { _ := pending.cancel() }
    frame.dispose()
    << 0
}
