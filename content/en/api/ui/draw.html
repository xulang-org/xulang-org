counts: 100 fns, 223 methods, 23 types, 50 consts, 0 ifaces

Explicit Context / Frame drawing with paths, gradients, text, images, DisplayList, Surface and caching. Panning is disabled by default; set_pan_enabled(1) enables the camera. Frame.scale uses factors; package-level scale uses percentages. Balance push / pop and retain resources until present / freeze / snapshot accepts the recording.

Usage and lifetimes

Release examples (complete projects and source downloads): draw_gallery

Drawing and game engines

This is a desktop 2D drawing layer suitable for tools, editors and modest prototypes. Its platform backends use CoreGraphics, Cairo and GDI+. The public API does not provide a game-engine GPU pipeline, shaders, draw batching or a display-synchronized frame-pacing contract. DisplayList and surface caches help reuse work, but are not evidence of a game frame-time budget; measure the intended workload on each target before committing to it.

fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>

        mut index := -1

        if index < 0 { -> }
        if event.kind == ui.EV_CANVAS_READY {

        input := ui.input_of(&event) or >>
        if input.generation != windows[index].generation || windows[index].closed != 0 { -> }
        if input.kind == ui.EV_POINTER_UP && input.button == ui.BUTTON_LEFT {

        if input.key == ui.KEY_ESCAPE { demo.quit() or >> }

// The popup is an overlay, so route its pointer sequence before normal tree hits.
fn (m $&Model) dispatch_input(view $&widgets.View, event &ui.InputEvent)[]widgets.Action{
    was_open := m.select_ctrl.is_open()
    pointer := event.kind == ui.EV_POINTER_DOWN || event.kind == ui.EV_POINTER_UP ||
    event.kind == ui.EV_POINTER_MOVE || event.kind == ui.EV_POINTER_SCROLL
    if pointer &&(was_open || m.select_popup_armed) {
        local := view.local_input(41, event) or >>
        styles := view.styles_for(41) or >>
        pop := m.select_ctrl.popup_bounds(&styles)
        if pop.contains(local.x, local.y) || m.select_popup_armed{
            if event.kind == ui.EV_POINTER_DOWN && event.button == ui.BUTTON_LEFT{m.select_popup_armed = true}
            control := $&m.select_ctrl
            action := control.input(&local) or >>
            view.set_pointer_cursor(ui.CURSOR_HAND) or >>
            if event.kind == ui.EV_POINTER_UP{m.select_popup_armed = false}
            view.mark_paint(41) or >>
            view.mark_paint(1) or >>
            <<[action]
        }
    }
    if event.kind == ui.EV_POINTER_CANCEL{m.select_popup_armed = false}
    actions := widgets.dispatch(view, event, m) or >>
    if was_open || m.select_ctrl.is_open() {view.mark_paint(1) or >> }
    << actions
}
fn (m $&Model) dispatch_event(view $&widgets.View, event &ui.Event)[]widgets.Action{

        input := ui.input_of(event) or >>
        << m.dispatch_input(view, &input) or >>
    }
    was_open := m.select_ctrl.is_open()
    actions := widgets.dispatch_event(view, event, m) or >>
    m.select_popup_armed = false
    if was_open && !m.select_ctrl.is_open() {view.mark_paint(1) or >> }
    << actions
}

        kind == widgets.CHART_BULLET || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_PARETO

            name = "Build"

        }
        $series := chart.chart_series(si + 1, name, kind)
        series.smooth = kind == widgets.CHART_LINE || kind == widgets.CHART_AREA || kind == widgets.CHART_STACKED_AREA ||
            kind == widgets.CHART_SMOOTH_LINE || kind == widgets.CHART_SMOOTH_AREA
        $count := 12 - revision % 2
        if kind == widgets.CHART_SCATTER{count = 64}
        else if kind == widgets.CHART_BUBBLE{count = 16}
        else if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||

        else if kind == widgets.CHART_WATERFALL{count = 7}
        for i in 0:count{
            x := f64(i + 1)
            phase := f64(revision) * 0.9 + f64(si) * 1.7
            $y := 48.0 + f64(si) * 20.0 + math.sin(x * 0.55 + phase) * 20.0 + math.cos(x * 0.17 + phase) * 7.0
            if kind == widgets.CHART_FUNNEL || kind == widgets.CHART_PARETO || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                y = f64((count - i) * 22 + (revision % 3) * 4)
            } else if kind == widgets.CHART_GAUGE || kind == widgets.CHART_POLAR_BAR{
                y = math.max(12.0, math.min(96.0, 34.0 + f64(i) * 18.0 + math.sin(phase + x) * 14.0))

            $point := chart.chart_point(i + 1, x, y)

                }
                low := math.min(y, close) - 8.0 - math.abs(math.cos(x + phase)) * 6.0
                high := math.max(y, close) + 8.0 + math.abs(math.sin(x + phase)) * 6.0
                point = chart.chart_range_point(i + 1, x, y, low, high, close)
            } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                point.size = 1.0
            }

                } else if kind == widgets.CHART_GAUGE{
                    point.label =["CPU", "Memory", "Cache"][i]

                    point.label =["Core 0", "Core 1", "Core 2", "Core 3"][i]

                    point.label = "Total"
                }
            } else {
                point.label = "样本 " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{

                    point.label = "总计"
                }
            }
            series.points.push(point)
        }
        result.push(series)
    }
