#net/http

Xu {{xu_version}} API

import: #net/http

counts: 21 fns, 49 methods, 9 types, 0 consts, 0 ifaces

Servers dispatch complete requests concurrently with bounded connection intake, buffers and timeouts. Handlers may overlap; synchronize shared mutable state and transfer request ownership explicitly. Use server_options() for defaults and serve_with_options to configure limits. Short response writes are retried. Idle or incomplete requests do not occupy a handler indefinitely.

// Go:
//   http.ListenAndServe(":8080", http.FileServer(http.Dir(".")))
// Xu:
//   http.listen_and_serve("127.0.0.1:8080", ".")
//
// 参数：要对外的目录。默认本示例的 www/（从仓库根运行）。

fn main() int {
    $root := "examples/12_file_server/www"
    av := args()
    if av.len >= 2 {
        root = av[1] + ""
    }
    println("FileServer " + root)
    println("http://127.0.0.1:8080")
    http.listen_and_serve("127.0.0.1:8080", root) or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

// 写法 C — 请求上下文（Go Handler / HandlerFunc）。
//
// Go:
//   func hello(w http.ResponseWriter, r *http.Request) {
//       fmt.Fprint(w, "hello")
//   }
//   http.ListenAndServe(":8080", http.HandlerFunc(hello))
//
// Xu: w 和 r 收成一个 Context。

fn app(c http.Context) {
    p := c.path()
    if p == "/" || p == "/index.html" {
        page := "<!doctype html><html><head><meta charset=\"utf-8\"><title>Xu</title></head>" +
            "<body><h1>Context</h1>" +
            "<p><a href=\"/hello\">/hello</a> · <a href=\"/health\">/health</a></p>" +
            "<p>POST /echo</p></body></html>"
        c.html(200, page) or (err) { << }
        <<
    }
    if p == "/hello" {
        c.text(200, "hello, xu\n") or (err) { << }
        <<
    }
    if p == "/health" {
        c.text(200, "ok\n") or (err) { << }
        <<
    }
    if p == "/echo" {
        m := c.method()
        if m != "POST" && m != "PUT" {
            c.text(405, "method not allowed\n") or (err) { << }
            <<
        }
        b := c.body()
        c.text(200, b) or (err) { << }
        <<
    }
    miss := "not found: " + p + "\n"
    c.text(404, miss) or (err) { << }
}

fn main() int {
    println("http://127.0.0.1:18081")
    http.listen_and_serve_handler("127.0.0.1:18081", app) or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

#net/http

// Xu Design System & VI Specification 官方本地 HTTP 服务
// 运行指令：
//   ./xu_design_server.xin [--addr 127.0.0.1:8088] [--root <www-dir>]
//   位置参数仍兼容旧形式：<port> <www-dir>。

fn main() int {
    $addr := "127.0.0.1:8088"
    $root := "examples/16_ui_webview/xu_design_system/www"

    av := args()
    $i := 1
    while i < av.len {
        arg := av[i] + ""
        if arg == "--help" || arg == "-h" {
            println("Xu Design System HTTP server")
            println("usage: xu_design_server [--addr HOST:PORT] [--root DIR]")
            println("legacy: xu_design_server PORT DIR")
            << 0
        }
        if arg == "--addr" {
            if i + 1 >= av.len {
                println("missing value for --addr")
                << 2
            }
            addr = av[i + 1] + ""
            i = i + 2
            ->
        }
        if arg == "--root" {
            if i + 1 >= av.len {
                println("missing value for --root")
                << 2
            }
            root = av[i + 1] + ""
            i = i + 2
            ->
        }
        // Preserve the original positional interface: port, then root.
        if i == 1 {
            addr = "127.0.0.1:" + arg
        } else if i == 2 {
            root = arg
        } else {
            println("unknown argument: " + arg)
            << 2
        }
        i = i + 1
    }

    println("==================================================================")
    println("    __  __  _   _  ")
    println("    \\ \\/ / | | | |   ⚡ Xu Official HTTP Server (Design System)")
    println("     \\  /  | | | |   🔥 Rock-Solid Ownership · Supersonic AOT")
    println("     /  \\  | |_| |   🌐 Serving: " + root)
    println("    /_/\\_\\  \\___/    🔗 Local URL: http://" + addr + "/")
    println("==================================================================")
    println("Press Ctrl+C to stop server.\n")

    http.listen_and_serve(addr, root) or (err) {
        println("Server error: " + err.msg())
        << 1
    }
    << 0
}
