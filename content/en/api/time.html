fn (m $&Model) work(screen $&widgets.Screen,now time.Instant) int {
    if !m.running || now.before(m.next_work) { << 0 }
    // Each progress update follows an actual bounded work batch. The pause
    // represents an application work queue and never changes the percentage.
    for i in 0:5000 { m.checksum = (m.checksum + i * (m.completed + 1)) % 1000003 }
    m.completed = m.completed + 1
    _ := screen.set_task_progress(m.task,m.task_id,f64(m.completed) / 48.0) or >>
    m.next_work = now.add(time.milliseconds(35) or >>) or >>
    if m.fail_next && m.completed == 18 {
        _ := screen.complete_task(m.task,m.task_id,widgets.TASK_ERROR) or >>
        m.running = false
        m.fail_next = false
        screen.set_text(m.message,"Example failure. Retry starts a new task; stale progress is rejected.") or >>
    } else if m.completed == 48 {
        _ := screen.complete_task(m.task,m.task_id,widgets.TASK_SUCCESS) or >>
        m.running = false
        screen.set_text(m.message,"48/48 work batches completed · checksum " + str(m.checksum)) or >>
    }
    << 0
}

fn next_event(events chan ui.Event,deadline time.Instant) ui.Event {
    if deadline.is_zero() { << <-events }
    delay := deadline.since(time.instant())
    if delay.nanoseconds() <= 0 { << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0} }
    timer := delay.timer()
    select {
        event := <-events { timer.stop()
            << event }
        _ := <-timer.c { timer.stop() }
    }
    << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0}
}
fn scene(events chan ui.Event) int {
    ready := demo.ready(events) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    $screen := widgets.screen(&ready) or >>
    screen.resize(1000.0,820.0) or >>
    screen.set_min_size(900,720) or >>
    $model := make_model($&screen) or (err) { screen.dispose()
        >> err }
    model.layout($&screen) or >>
    _ := screen.render() or >>
    while screen.is_open() {
        now := time.instant()
        model.work($&screen,now) or >>
        model.layout($&screen) or >>
        $deadline := screen.advance(now) or (err) { >> errors.new("gallery animation: " + err.msg()) }
        for action in screen.take_actions() { model.update($&screen,action) or >> }
        _ := screen.render() or (err) { >> errors.new("gallery render: " + err.msg()) }
        if model.running && (deadline.is_zero() || model.next_work.before(deadline)) { deadline = model.next_work }
        event := next_event(events,deadline) or >>
        $actions := screen.event(&event) or >>
        $count := 0
        while screen.is_open() && count < 128 {
            pending := ui.poll_chan(events)
            if pending.kind == ui.EV_NONE { <- }
            more := screen.event(&pending) or >>
            for action in more { actions.push(action) }
            count += 1
        }
        for action in actions { model.update($&screen,action) or >> }
    }
    screen.dispose()
    << 0
}

fn (m $&Model) tick(view $&widgets.View,now time.Instant) int {
    m.clock = now
    reduced := m.reduced_motion
    for id in 1:m.transitions.len {
        motion := $&m.transitions[id]
        if reduced && motion.is_animating() { motion.finish()
            view.mark_paint(id) or >> }
        else if motion.advance(now) or >> { view.mark_paint(id) or >> }
    }
    $selected := 0.0
    if m.toggle.value() { selected = 1.0 }
    switch_motion := $&m.switch_motion
    before_switch := switch_motion.value()
    switch_motion.retarget(selected,now,time.milliseconds(180) or >>,reduced) or >>
    if before_switch != switch_motion.value() { view.mark_paint(32) or >> }
    $opened := 0.0
    if m.select_ctrl.is_open() { opened = 1.0 }
    popup_motion := $&m.popup_motion
    before_popup := popup_motion.value()
    $popup_duration := time.milliseconds(130) or >>
    if opened == 0.0 { popup_duration = time.milliseconds(95) or >> }
    popup_motion.retarget(opened,now,popup_duration,reduced) or >>
    if before_popup != popup_motion.value() { view.mark_paint(1) or >> }
    input := $&m.input
    if input.tick(now,reduced) or >> { view.mark_paint(12) or >> }
    invalid := $&m.invalid
    if invalid.tick(now,reduced) or >> { view.mark_paint(13) or >> }
    read_only := $&m.read_only
    if read_only.tick(now,reduced) or >> { view.mark_paint(14) or >> }
    area := $&m.area
    if area.tick(now,reduced) or >> { view.mark_paint(15) or >> }
    << 0
}

// The popup is an overlay, so route its pointer sequence before normal tree hits.
fn (m $&Model) dispatch_input(view $&widgets.View,event &ui.InputEvent) []widgets.Action {
    was_open := m.select_ctrl.is_open()
    pointer := event.kind == ui.EV_POINTER_DOWN || event.kind == ui.EV_POINTER_UP ||
        event.kind == ui.EV_POINTER_MOVE || event.kind == ui.EV_POINTER_SCROLL
    if pointer && (was_open || m.select_popup_armed) {
        local := view.local_input(41,event) or >>
        styles := view.styles_for(41) or >>
        pop := m.select_ctrl.popup_bounds(&styles)
        if pop.contains(local.x,local.y) || m.select_popup_armed {
            if event.kind == ui.EV_POINTER_DOWN && event.button == ui.BUTTON_LEFT { m.select_popup_armed = true }
            control := $&m.select_ctrl
            action := control.input(&local) or >>
            view.set_pointer_cursor(ui.CURSOR_HAND) or >>
            if event.kind == ui.EV_POINTER_UP { m.select_popup_armed = false }
            view.mark_paint(41) or >>
            view.mark_paint(1) or >>
            << [action]
        }
    }
    if event.kind == ui.EV_POINTER_CANCEL { m.select_popup_armed = false }
    actions := widgets.dispatch(view,event,m) or >>
    if was_open || m.select_ctrl.is_open() { view.mark_paint(1) or >> }
    << actions
}
fn (m $&Model) dispatch_event(view $&widgets.View,event &ui.Event) []widgets.Action {
    if ui.is_input_event(event.kind) {
        input := ui.input_of(event) or >>
        << m.dispatch_input(view,&input) or >>
    }
    was_open := m.select_ctrl.is_open()
    actions := widgets.dispatch_event(view,event,m) or >>
    m.select_popup_armed = false
    if was_open && !m.select_ctrl.is_open() { view.mark_paint(1) or >> }
    << actions
}

fn layout_buttons(s $&widgets.Screen,page &Page,content f64,row f64,scale f64) f64 {
    stepper_width := math.min(content,160.0 * scale)
    label_height := 24.0 * scale
    inspector_height := label_height + 8.0 + row + 12.0
    $cy := 24.0
    // Keep each direction pair together when it fits, then wrap by the native
    // measured size rather than stretching every example across the preview.
    for group in 0:2 {
        $x := 24.0
        $line_height := 0.0
        for i in (group * 2):(group * 2 + 2) {
            size := s.measure_button(page.ids[i]) or >>
            width := math.min(content,math.ceil(size.width))
            height := math.max(row,math.ceil(size.height))
            column := math.max(width,stepper_width)
            if x > 24.0 && x + column > 24.0 + content {
                x = 24.0
                cy += line_height + 12.0
                line_height = 0.0
            }
            s.place(page.ids[13 + i],x,cy,column,label_height) or >>
            s.place(page.ids[9 + i],x,cy + label_height + 8.0,stepper_width,row) or >>
            s.place(page.ids[i],x,cy + inspector_height,width,height) or >>
            x += column + 16.0
            line_height = math.max(line_height,inspector_height + height)
        }
        cy += line_height + 20.0
    }
    $x := 24.0
    $line_height := 0.0
    for i in 4:9 {
        size := s.measure_button(page.ids[i]) or >>
        width := math.min(content,math.ceil(size.width))
        height := math.max(row,math.ceil(size.height))
        if x > 24.0 && x + width > 24.0 + content {
            x = 24.0
            cy += line_height + 12.0
            line_height = 0.0
        }
        s.place(page.ids[i],x,cy,width,height) or >>
        x += width + 12.0
        line_height = math.max(line_height,height)
    }
    cy += line_height + 20.0
    << cy - 24.0
}

fn (m $&Workshop) work(s $&widgets.Screen,now time.Instant) int {

    for i in 0:5000 { m.checksum = (m.checksum + i * (m.completed + 1)) % 1000003 }

    else if m.completed == 48 { _ := s.complete_task(id,m.revision,widgets.TASK_SUCCESS) or >>
        m.running = false }
    << 0
}

fn chart_data(kind int,revision int) []widgets.ChartSeries {
    $result := []widgets.ChartSeries{}
    $series_count := 2
    if kind == widgets.CHART_DONUT { series_count = 1 }
    for si in 0:series_count {
        $name := "Build 构建"
        if si == 1 { name = "Index 索引" }
        if kind == widgets.CHART_DONUT { name = "Workload 工作量" }
        $series := widgets.chart_series(si + 1,name,kind)
        series.smooth = kind == widgets.CHART_LINE || kind == widgets.CHART_AREA
        $count := 12 - revision % 2
        if kind == widgets.CHART_SCATTER { count = 64 }
        if kind == widgets.CHART_DONUT { count = 5 }
        for i in 0:count {
            x := f64(i + 1)
            phase := f64(revision) * 0.9 + f64(si) * 1.7
            y := 48.0 + f64(si) * 20.0 + math.sin(x * 0.55 + phase) * 20.0 + math.cos(x * 0.17 + phase) * 7.0
            $point := widgets.chart_point(i + 1,x,y)
            point.label = "Sample 样本 " + str(i + 1)
            if kind == widgets.CHART_DONUT { point.label = ["Compile 编译","Index 索引","Render 绘制","I/O 输入输出","Other 其他"][i] }
            series.points.push(point)
        }
        result.push(series)
    }
    << result
}

fn next_event(events chan ui.Event,deadline time.Instant) ui.Event {
    if deadline.is_zero() { << <-events }
    delay := deadline.since(time.instant())
    if delay.nanoseconds() <= 0 { << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0} }
    timer := delay.timer()
    select { event := <-events { timer.stop()
            << event }
        _ := <-timer.c { timer.stop() } }
    << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0}
}

fn (m &Workshop) interaction_layout(s $&widgets.Screen,row f64,scale f64,content f64) f64 {
    position := m.selected
    pages := &m.pages
    page := &pages[position]
    id := page.ids[0]
    if m.selected == 45 {
        width := math.min(content,380.0 * scale)
        label_height := 28.0 * scale
        input_height := math.max(row,48.0 * s.styles().density)
        input_y := 24.0 + label_height + 8.0
        s.place(page.ids[2],24.0,24.0,width,label_height) or >>
        s.place(id,24.0,input_y,width,input_height) or >>
        s.place(page.ids[1],24.0,input_y + input_height + 16.0,content,row) or >>
        s.place(page.ids[3],24.0,input_y + input_height + row + 36.0,content,100.0 * scale) or >>
        << label_height + input_height + row + 160.0 * scale
    }
    if m.selected == 40 {
        $width := content
        $height := 104.0*scale
        if m.range_vertical { width = math.min(content,160.0*scale)
            height = 300.0*scale }
        s.place(page.ids[1],24.0,24.0,content,32.0*scale) or >>
        s.place(id,24.0,64.0*scale,width,height) or >>
        s.place(page.ids[2],24.0,80.0*scale+height,content,64.0*scale) or >>
        << height+144.0*scale
    }
    if m.selected == 41 {
        height := 320.0*scale
        s.place(id,24.0,24.0,content,height) or >>
        boxes := s.split_layout(id) or >>
        for pane in 0:2 {
            $width := boxes.first.width
            $height := boxes.first.height
            if pane == 1 { width = boxes.second.width
                height = boxes.second.height }
            inner := math.max(1.0,width-32.0)
            caption_height := math.max(1.0,math.min(100.0*scale,height-row-48.0))
            s.place(page.ids[3+pane],16.0,16.0,inner,caption_height) or >>
            s.set_visible(page.ids[5+pane],height >= row+32.0) or >>
            s.place(page.ids[5+pane],16.0,math.max(16.0,height-row-16.0),inner,row) or >>
        }
        << height
    }
    if m.selected >= 42 && m.selected <= 44 {
        $width := math.min(content,240.0*scale)
        $height := row
        if m.selected == 43 { width = content
            height = 180.0*scale }
        s.place(id,24.0,24.0,width,height) or >>
        s.place(page.ids[1],24.0,height+44.0*scale,content,44.0*scale) or >>
        s.place(page.ids[2],24.0,height+100.0*scale,content,80.0*scale) or >>
        << height+180.0*scale
    }
    << row
}

fn scene(events chan ui.Event) int {
    ready := demo.ready(events) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    $screen := widgets.screen(&ready) or >>
    screen.resize(1180.0,900.0) or >>
    screen.set_min_size(920,780) or >>
    $model := create($&screen) or (err) { screen.dispose()
        >> err }
    model.apply_locale($&screen) or >>
    screen.set_text(model.title,"XUUI  /  Field") or >>
    model.configure($&screen) or >>
    model.layout($&screen) or >>
    while screen.is_open() {
        now := time.instant()
        model.work($&screen,now) or >>
        model.layout($&screen) or >>
        $deadline := screen.advance(now) or >>
        for action in screen.take_actions() { model.update($&screen,action) or >> }
        _ := screen.render() or >>
        if model.running && (deadline.is_zero() || model.deadline.before(deadline)) { deadline = model.deadline }
        event := next_event(events,deadline) or >>
        $actions := screen.event(&event) or >>
        $count := 0
        while screen.is_open() && count < 128 { pending := ui.poll_chan(events)
            if pending.kind == ui.EV_NONE { <- }
            more := screen.event(&pending) or >>
            for action in more { actions.push(action) }
            count += 1 }
        for action in actions { model.update($&screen,action) or >> }
    }
    screen.dispose()
    << 0
}
