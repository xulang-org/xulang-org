fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    seed := "你好，Xu 👩‍💻 · café"
    caret := ui.TextInputRect{x: 44.0,y: 180.0,width: 2.0,height: 36.0}
    options := ui.text_input_options(&seed,caret)
    mut input := ui.start_text_input(identity.canvas_id,identity.generation,&options) or >>
    mut frame := context.new_frame(4096)
    paint(&context,mut &frame,&input) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        mut dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind >= ui.EV_TEXT_COMMIT && event.kind <= ui.EV_TEXT_SURROUNDING_DELETE {
            edit := ui.text_edit_of(&event) or >>
            if edit.input.id == input.id {
                if event.kind == ui.EV_TEXT_INPUT_ENDED || event.kind == ui.EV_TEXT_INPUT_FAILED { <- }
                dirty = true
            }
        }
        if ui.is_input_event(event.kind) {
            pointer := ui.input_of(&event) or >>
            if pointer.generation != identity.generation { -> }
            if pointer.kind == ui.EV_KEY_DOWN && pointer.key == ui.KEY_ESCAPE { demo.quit() or >> }
            bounds := layout.Rect{x: 28.0,y: 160.0,width: 790.0,height: 86.0}
            if pointer.kind == ui.EV_POINTER_DOWN && pointer.button == ui.BUTTON_LEFT && bounds.contains(pointer.x,pointer.y) {
                state := input.snapshot() or >>
                // The native editor owns preedit; do not overwrite an active composition.
                if !state.composing {
                    hit := input.hit_test(state.revision,pointer.x,pointer.y) or >>
                    index := hit.byte_index()
                    _ := input.update(state.revision,&state.value,index,index,&caret) or >>
                    dirty = true
                }
            }
        }
        if dirty { paint(&context,mut &frame,&input) or >> }
    }
    frame.dispose()
    input.stop() or (err) { << 0 }
    << 0
}

fn layout_buttons(s $&widgets.Screen, page &Page, content f64, row f64, scale f64) f64 {
    stepper_width := math.min(content, 160.0 * scale)
    label_height := 24.0 * scale
    inspector_height := label_height + 8.0 + row + 12.0
    $cy := 24.0

    // measured size rather than stretching every example across the preview.
    for group in 0:2 {

                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "Total"
                }
            } else {
                point.label = "样本 " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                    point.label =["编译", "索引", "绘制", "输入输出", "其他"][i]
                } else if kind == widgets.CHART_RADAR{
                    point.label =["吞吐量", "延迟", "内存", "启动", "缓存", "体积"][i]
                } else if kind == widgets.CHART_GAUGE{
                    point.label =["处理器", "内存", "缓存"][i]
                } else if kind == widgets.CHART_POLAR_BAR{
                    point.label =["核心 0", "核心 1", "核心 2", "核心 3"][i]
                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "总计"
                }
            }
            series.points.push(point)
        }
        result.push(series)
