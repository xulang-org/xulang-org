fn main() int {
    println("=== NATS Cloud-Native Messaging Standalone Integration Test ===")
    addr := "127.0.0.1:4222"
    println("Connecting to NATS at " + addr + "...")

    mut nc := nats.connect(addr + "") or (err) {
        println("Connection to NATS failed: " + err.msg())
        << 1
    }

    ping_ok := nc.ping() or (err) {
        println("Ping failed: " + err.msg())
        << 1
    }
    println("1. NATS Ping response: " + str(ping_ok))

    // 2. Subscribe to subject
    topic := "xu.events"
    sub := nc.subscribe(topic + "") or (err) {
        println("Subscribe failed: " + err.msg())
        << 1
    }
    println("2. Subscribed to subject: " + sub.subject + " (sid=" + sub.sid + ")")

    // 3. Publish a message
    payload := "Hello NATS from Xu Programming Language!"
    nc.publish(topic + "", payload + "") or (err) {
        println("Publish failed: " + err.msg())
        << 1
    }
    println("3. Published payload: " + payload)

    // 4. Receive the message
    msg := nc.next_msg() or (err) {
        println("Receive failed: " + err.msg())
        << 1
    }
    println("4. Successfully received message:")
    println("   - Subject: " + msg.subject)
    println("   - Sid: " + msg.sid)
    println("   - Data: " + msg.data)

    // 5. Test Queue Group (load-balanced worker queue)
    q_sub := nc.queue_subscribe("xu.tasks", "workers") or (err) {
        println("Queue subscribe failed: " + err.msg())
        << 1
    }
    println("5. Queue Group subscribed: xu.tasks -> workers (sid=" + q_sub.sid + ")")

    nc.publish("xu.tasks", "task_payload_123") or (err) {
        << 1
    }
    q_msg := nc.next_msg() or (err) {
        << 1
    }
    println("6. Queue Group received task: " + q_msg.data)

    nc.unsubscribe(sub) or (err) {
        << 1
    }
    nc.unsubscribe(q_sub) or (err) {
        << 1
    }
    nc.close()

    // 7. NATS Connection Pool Deep Verification
    println("7. Testing NATS Connection Pool with Reuse & Concurrency...")
    pool := nats.open_pool("127.0.0.1:4222", 5)
    mut c1 := pool.get() or (err) {
        println("Pool get c1 failed: " + err.msg())
        << 1
    }
    fd_first := c1.fd
    c1_ok := c1.ping() or (err) {
        << 1
    }
    println("   Pool c1 ping: " + str(c1_ok) + " (fd=" + str(fd_first) + ")")
    pool.put(c1)

    // Verify Pool Connection REUSE
    mut c2 := pool.get() or (err) {
        println("Pool get c2 failed: " + err.msg())
        << 1
    }
    fd_second := c2.fd
    if fd_first == fd_second {
        println("   [PASS] NATS Pool Connection REUSE verified! (fd1 == fd2: " + str(fd_first) + ")")
    } else {
        println("   [WARN] fd1 != fd2 (fd1=" + str(fd_first) + ", fd2=" + str(fd_second) + ")")
    }

    // Concurrent borrow another connection
    mut c3 := pool.get() or (err) {
        println("Pool get c3 failed: " + err.msg())
        << 1
    }
    fd_third := c3.fd
    if fd_second != fd_third {
        println("   [PASS] NATS Pool Multi-connection concurrency verified! (fd2=" + str(fd_second) + " != fd3=" + str(fd_third) + ")")
    }

    // Pool Direct Publish
    pool.publish("pool.notify", "notification from nats pool") or (err) {
        << 1
    }
    println("   [PASS] NATS Pool direct publish succeeded.")

    pool.put(c2)
    pool.put(c3)
    pool.close()
    println("   [PASS] NATS Pool closed cleanly.")

    println("=== NATS Tests Completed Successfully! ===")
    << 0
}
