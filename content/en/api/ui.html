fn route(request webview.InvokeRequest) string {
    if request.name == "slow" { sleep_ns(300000000) }
    if request.name == "wait" { sleep_ns(1500000000) }
    if request.name == "native" {
        target := ui.CanvasEvent{window_id: request.window_id,canvas_id: request.window_id,generation: request.generation}
        options := ui.RequestOptions{timeout_ms: 2000}
        script := "document.title"
        result := ui.webview_request_wait(&target,ui.REQUEST_WEBVIEW_EVAL,&script,&options) or (err) { << "failed: " + err.msg() }
        if result.status != ui.REQUEST_COMPLETED { << "status: " + str(result.status) }
        << result.value + ""
    }
    << request.payload + ""
}
fn ipc(emit_ch chan string,ech chan ui.Event) {

    }
}

fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    mut app := demo.windows(ech)
    first := ui.Window{id: identity.window_id,generation: identity.generation,title: "Xu UI / Window 1",width: 560,height: 340,content_kind: ui.CONTENT_DRAW,content_backend_id: identity.canvas_id,closed: 0}
    mut windows := []ui.Window{}
    windows.push(first)
    mut counts := [0]
    paint(windows[0],0) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT { <- }
        mut index := -1
        for i in 0:windows.len { if windows[i].id == event.window_id { index = i } }
        if index < 0 { -> }
        if event.kind == ui.EV_CANVAS_READY {
            window := windows[index]
            window.content_command("input_enabled\t1") or >>
            window.content_command("pan_enabled\t0") or >>
            println("Canvas ready: " + str(event.window_id))
            paint(window,counts[index]) or >> }
        if event.kind == ui.EV_CANVAS_METRICS { paint(windows[index],counts[index]) or >> }
        if event.kind == ui.EV_CLOSED { windows[index].closed = 1 }
        if event.kind == ui.EV_WINDOW_FAILED { windows[index].closed = 1
            eprintln("Native window creation failed: " + event.menu_id) }
        if !ui.is_input_event(event.kind) { -> }
        input := ui.input_of(&event) or >>
        if input.generation != windows[index].generation || windows[index].closed != 0 { -> }
        if input.kind == ui.EV_POINTER_UP && input.button == ui.BUTTON_LEFT {
            counts[index] = counts[index] + 1
            println("Window " + str(event.window_id) + " count: " + str(counts[index]))
            paint(windows[index],counts[index]) or >>
        }
        if input.kind != ui.EV_KEY_DOWN || input.repeat != 0 { -> }
        if input.key == ui.KEY_ESCAPE { demo.quit() or >> }
        if input.key == 78 || input.key == 110 {
            // At most six demo windows; the host's own limit is independent.
            mut live := 0
            for window in windows { if window.closed == 0 { live = live + 1 } }
            if live < 6 {
                window := app.open_content_window(ui.WindowOpts{title: "Xu UI / Additional Window",width: 560,height: 340},ui.CONTENT_DRAW) or >>

                counts.push(0)
            }

            mut window := windows[index]
            window.request_close() or >>
            windows[index] = window
        }
        if input.key == 82 || input.key == 114 {
            mut window := windows[index]
            window.resize(660,400) or >>
            windows[index] = window
        }
        if input.key == 70 || input.key == 102 {
            for window in windows { if window.closed == 0 { window.focus() or >>

        }
    }
    << 0

fn watch_shell(shell_ch chan ui.Event) {
    while true {
        ev := <-shell_ch
        if ev.kind == ui.EV_EXIT || ev.kind == ui.EV_CLOSED {

            println("shell event loaded")
        } else if ev.kind == ui.EV_LOAD_FAILED {
            println("shell event load_failed")
        } else if ev.kind == ui.EV_READY {
            println("shell event ready")
        }
    }
}

fn tick_loop(emit_ch chan string) {
    mut n := 0
    while true {
        // Short sleeps so a blocked/mis-scheduled task cannot sit on a 2s nanosleep
        // that starves sibling work if threading is ever wrong. Emit only — no
        // native_eval/title on this loop (those jam the AppKit main thread).
        sleep_ns(200000000)
        n = n + 1
        if n % 10 == 0 {
            webview.bus_emit(emit_ch, "tick", str(n / 10))
        }
    }
}

    while true {

// The popup is an overlay, so route its pointer sequence before normal tree hits.
fn (m $&Model) dispatch_input(view $&widgets.View, event &ui.InputEvent)[]widgets.Action{
    was_open := m.select_ctrl.is_open()
    pointer := event.kind == ui.EV_POINTER_DOWN || event.kind == ui.EV_POINTER_UP ||
    event.kind == ui.EV_POINTER_MOVE || event.kind == ui.EV_POINTER_SCROLL
    if pointer &&(was_open || m.select_popup_armed) {
        local := view.local_input(41, event) or >>
        styles := view.styles_for(41) or >>
        pop := m.select_ctrl.popup_bounds(&styles)
        if pop.contains(local.x, local.y) || m.select_popup_armed{
            if event.kind == ui.EV_POINTER_DOWN && event.button == ui.BUTTON_LEFT{m.select_popup_armed = true}
            control := $&m.select_ctrl
            action := control.input(&local) or >>
            view.set_pointer_cursor(ui.CURSOR_HAND) or >>
            if event.kind == ui.EV_POINTER_UP{m.select_popup_armed = false}
            view.mark_paint(41) or >>
            view.mark_paint(1) or >>
            <<[action]
        }
    }
    if event.kind == ui.EV_POINTER_CANCEL{m.select_popup_armed = false}
    actions := widgets.dispatch(view, event, m) or >>
    if was_open || m.select_ctrl.is_open() {view.mark_paint(1) or >> }
    << actions
}

    m.select_popup_armed = false
    if was_open && !m.select_ctrl.is_open() {view.mark_paint(1) or >> }
    << actions

fn chart_data_locale(kind int, revision int, english bool)[]chart.ChartSeries{
    $result :=[]chart.ChartSeries{}
    $series_count := 2
    single := kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
        kind == widgets.CHART_GAUGE || kind == widgets.CHART_HISTOGRAM || kind == widgets.CHART_BOXPLOT ||
        kind == widgets.CHART_CANDLESTICK || kind == widgets.CHART_WATERFALL || kind == widgets.CHART_FUNNEL ||
        (kind >= widgets.CHART_POLAR_BAR && kind <= widgets.CHART_SANKEY) || kind == widgets.CHART_VIOLIN ||
        kind == widgets.CHART_BULLET || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_PARETO
    if single{series_count = 1}
    if kind == widgets.CHART_HEATMAP{series_count = 3}
    for si in 0:series_count{
        $name := "构建"
        if si == 1 {name = "索引"}
        if si == 2 {name = "绘制"}
        if single{name = "工作量"}
        if english {
            name = "Build"
            if si == 1 {name = "Index"}
            if si == 2 {name = "Render"}
            if single{name = "Workload"}
        }
        $series := chart.chart_series(si + 1, name, kind)
        series.smooth = kind == widgets.CHART_LINE || kind == widgets.CHART_AREA || kind == widgets.CHART_STACKED_AREA ||
            kind == widgets.CHART_SMOOTH_LINE || kind == widgets.CHART_SMOOTH_AREA
        $count := 12 - revision % 2
        if kind == widgets.CHART_SCATTER{count = 64}
        else if kind == widgets.CHART_BUBBLE{count = 16}
        else if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{count = 5}
        else if kind == widgets.CHART_RADAR || kind == widgets.CHART_PARETO{count = 6}
        else if kind == widgets.CHART_GAUGE{count = 3}
        else if kind == widgets.CHART_POLAR_BAR{count = 4}
        else if kind == widgets.CHART_HORIZONTAL_BAR || kind == widgets.CHART_HEATMAP || kind == widgets.CHART_BULLET ||
                kind == widgets.CHART_LOLLIPOP || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_VIOLIN{count = 8}
        else if kind == widgets.CHART_WATERFALL{count = 7}
        for i in 0:count{
            x := f64(i + 1)
            phase := f64(revision) * 0.9 + f64(si) * 1.7
            $y := 48.0 + f64(si) * 20.0 + math.sin(x * 0.55 + phase) * 20.0 + math.cos(x * 0.17 + phase) * 7.0
            if kind == widgets.CHART_FUNNEL || kind == widgets.CHART_PARETO || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                y = f64((count - i) * 22 + (revision % 3) * 4)
            } else if kind == widgets.CHART_GAUGE || kind == widgets.CHART_POLAR_BAR{
                y = math.max(12.0, math.min(96.0, 34.0 + f64(i) * 18.0 + math.sin(phase + x) * 14.0))
            } else if kind == widgets.CHART_WATERFALL{
                deltas :=[42.0, 18.0, 0.0 - 14.0, 24.0, 0.0 - 11.0, 16.0, 0.0]
                y = deltas[i] + math.sin(phase) * 3.0
                if i == count - 1 {y = 0.0}
            }
            $point := chart.chart_point(i + 1, x, y)
            if kind == widgets.CHART_BUBBLE{
                point = chart.chart_bubble_point(i + 1, x, y, 12.0 + math.abs(math.cos(x * 0.7 + phase)) * 36.0)
            } else if kind == widgets.CHART_BOXPLOT || kind == widgets.CHART_CANDLESTICK ||
                      kind == widgets.CHART_VIOLIN || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_BULLET{
                $close := y + math.sin(x * 0.9 + phase) * 12.0
                if (kind == widgets.CHART_BOXPLOT || kind == widgets.CHART_VIOLIN || kind == widgets.CHART_DUMBBELL || kind == widgets.CHART_BULLET) && close < y{
                    close = y + 10.0
                }
                low := math.min(y, close) - 8.0 - math.abs(math.cos(x + phase)) * 6.0
                high := math.max(y, close) + 8.0 + math.abs(math.sin(x + phase)) * 6.0
                point = chart.chart_range_point(i + 1, x, y, low, high, close)
            } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                point.size = 1.0
            }
            if english {
                point.label = "Sample " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                    point.label =["Compile", "Index", "Render", "I/O", "Other"][i]
                } else if kind == widgets.CHART_RADAR{
                    point.label =["Throughput", "Latency", "Memory", "Startup", "Cache", "Binary"][i]
                } else if kind == widgets.CHART_GAUGE{
                    point.label =["CPU", "Memory", "Cache"][i]
                } else if kind == widgets.CHART_POLAR_BAR{
                    point.label =["Core 0", "Core 1", "Core 2", "Core 3"][i]
                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "Total"
                }
            } else {
                point.label = "样本 " + str(i + 1)
                if kind == widgets.CHART_DONUT || kind == widgets.CHART_PIE || kind == widgets.CHART_ROSE ||
                   kind == widgets.CHART_FUNNEL || kind == widgets.CHART_TREEMAP || kind == widgets.CHART_SANKEY{
                    point.label =["编译", "索引", "绘制", "输入输出", "其他"][i]
                } else if kind == widgets.CHART_RADAR{
                    point.label =["吞吐量", "延迟", "内存", "启动", "缓存", "体积"][i]
                } else if kind == widgets.CHART_GAUGE{
                    point.label =["处理器", "内存", "缓存"][i]

                    point.label =["核心 0", "核心 1", "核心 2", "核心 3"][i]
                } else if kind == widgets.CHART_WATERFALL && i == count - 1 {
                    point.label = "总计"
                }
            }
            series.points.push(point)
        }
        result.push(series)
    }
    << result
}
