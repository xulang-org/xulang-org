import: #ui

Windows, events, structured menus, screens and asynchronous desktop services. One worker consumes App.event_ch; validate window_id / canvas_id / generation. Decode ev_request with request_of, match it with RequestResult.matches, and handle completed, cancelled, failed, unsupported and timed_out. Screen and canvas coordinates differ.

// Shaped window: frameless + transparent + PNG alpha mask (macOS / Linux / Windows).
// Chrome flags are written BEFORE App.run() so the first frame has no titlebar.
//   xu build examples/16ui_webview/shaped --target macos-arm64 --desktop -o /tmp/xu_shaped
//   /tmp/xu_shaped

fn main() int {
    $app := ui.new_app()
    // Pack diagnostics: shim sets XU_PACK_TRACE beside the exe.
    t := env_get("XU_PACK_TRACE")
    if t.len > 0 {
        _ := write_file(t, "shaped main begin\n") or (err) {
            << 0
        }
    }
    $win := app.new_window(ui.WindowOpts{title: "Xu Shaped", width: 320, height: 320})
    a0 := ui.native_set_frameless(1)
    a1 := ui.native_set_transparent(1)
    a2 := ui.native_set_window_mask("brand/xu_app_logo.png") or (err) {
        if t.len > 0 {
            _ := write_file(t, "mask fail: " + err.msg() + "\n") or (err2) {
                << 0
            }
        }
        println(err.msg())
        << 1
    }
    a3 := ui.native_set_shadow(1)
    if a0 + a1 + a2 + a3 < 0 {
        << 1
    }
    if t.len > 0 {
        _ := write_file(t, "shaped before app.run\n") or (err) {
            << 0
        }
    }
    println("shaped window — no titlebar; Esc or right-click Close; mask brand/xu_app_logo.png")
    app.run() or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

fn route(request webview.InvokeRequest) string {
    if request.name == "slow" { sleep_ns(300000000) }
    if request.name == "wait" { sleep_ns(1500000000) }
    if request.name == "native" {
        target := ui.CanvasEvent{window_id: request.window_id,canvas_id: request.window_id,generation: request.generation}
        options := ui.RequestOptions{timeout_ms: 2000}
        script := "document.title"
        result := ui.webview_request_wait(&target,ui.REQUEST_WEBVIEW_EVAL,&script,&options) or (err) { << "failed: " + err.msg() }
        if result.status != ui.REQUEST_COMPLETED { << "status: " + str(result.status) }
        << result.value + ""
    }
    << request.payload + ""
}
fn ipc(emit_ch chan string,ech chan ui.Event) {
    webview.bind_ipc_requests_www(":mem:",route,emit_ch,ech) or (err) { eprintln(err.msg())
        << }
}
fn events(ech chan ui.Event,emit_ch chan string) {
    while true { event := <-ech
        if event.kind == ui.EV_EXIT { emit_ch <- ""
            << }
        if event.kind == ui.EV_CLOSED { println("Window closed; pending document requests revoked") }
    }
}
fn run() int {
    ech := chan ui.Event{cap: 1024}
    emit_ch := chan string{cap: 64}
    $app := ui.new_app_events(ech)
    webview.stage_www_root(":mem:") or >>
    xu ipc(emit_ch,ech)
    xu events(ech,emit_ch)
    $window := app.new_window(ui.WindowOpts{title: "Xu UI / WebView Request Lifecycle",width: 880,height: 640})
    $browser := webview.attach($&window)
    browser.load_url(webview.app_scheme_url()) or >>
    << app.run() or >>
}
fn main() int { << run() or (err) { eprintln(err.msg())
    << 1 } }

fn paint(context &draw.Context,frame $&draw.Frame,hits int) int {
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,32.0,28.0,14.0,demo.accent(),"01 / FRAME RECORDER") or >>
    demo.text(context,frame,32.0,58.0,30.0,demo.ink(),"Draw a frame. Submit once.") or >>
    demo.text(context,frame,32.0,108.0,15.0,demo.muted(),"Click the teal circle. Resize the window. Escape closes it.") or >>
    frame.set_fill(demo.accent()) or >>
    frame.circle(112.0,228.0,52.0) or >>
    frame.stroke_rect(200.5,176.5,128.0,104.0,draw.rgb(240,180,91),2.0) or >>
    frame.push() or >>
    frame.clip_rect(376.0,168.0,216.0,120.0) or >>
    frame.translate(478.0,226.0) or >>
    frame.rotate(18.0) or >>
    frame.fill_rect(-100.0,-42.0,200.0,84.0,draw.rgb(92,127,203)) or >>
    frame.pop() or >>
    demo.text(context,frame,32.0,330.0,20.0,demo.ink(),"Circle activations: " + str(hits)) or >>
    demo.text(context,frame,32.0,370.0,13.0,demo.muted(),"Logical coordinates / state stack / clipping / one present") or >>
    frame.present() or >>
    println("Frame submitted: " + str(frame.command_count()) + " commands / " + str(frame.byte_len()) + " bytes")
    << 0
}
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    $frame := context.new_frame(4096)
    $hits := 0
    $pressed := false
    paint(&context,$&frame,hits) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT { <- }
        if event.window_id != identity.window_id { -> }
        if event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.kind == ui.EV_BLURRED { pressed = false }
        if event.kind == ui.EV_CANVAS_METRICS { paint(&context,$&frame,hits) or >> }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            if input.kind == ui.EV_KEY_DOWN && input.key == ui.KEY_ESCAPE { demo.quit() or >> }
            dx := input.x - 112.0
            dy := input.y - 228.0
            inside := dx * dx + dy * dy <= 2704.0
            if input.kind == ui.EV_POINTER_DOWN && input.button == ui.BUTTON_LEFT { pressed = inside }
            if input.kind == ui.EV_POINTER_CANCEL { pressed = false }
            if input.kind == ui.EV_POINTER_UP && input.button == ui.BUTTON_LEFT {
                if pressed && inside { hits = hits + 1
                    paint(&context,$&frame,hits) or >> }
                pressed = false
            }
        }
    }
    frame.dispose()
    << 0
}
fn worker(ech chan ui.Event) { scene(ech) or (err) { demo.fail(err.msg())
    << } }
fn launch() int {
    $app := demo.app("Xu UI / Frame",680,430) or >>
    xu worker(app.event_ch)
    << app.run() or >>
}
fn main() int { << launch() or (err) { eprintln(err.msg())
    << 1 } }
