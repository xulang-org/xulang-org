// MySQL/MariaDB：XU_MYSQL_DSN 可覆盖本地开发默认值。
// 驱动是普通 XU 包，通过 database/sql 接口使用，不依赖编译器内建。

fn main() int {
    mut dsn := os.getenv("XU_MYSQL_DSN")
    if dsn.len == 0 {
        dsn = "root:root@tcp(127.0.0.1:3306)/xu_demo?tls=false"
    }

    mut db := sql.open(mysql.driver(), dsn)
    db.ping() or (err) {
        println("MySQL ping failed")
        println(err.msg())
        << 1
    }

    db.exec("DROP TABLE IF EXISTS xu_people") or (err) {
        println(err.msg())
        << 1
    }
    db.exec("CREATE TABLE xu_people (id INTEGER PRIMARY KEY, name VARCHAR(100) NOT NULL, active BOOLEAN NOT NULL)") or (err) {
        println(err.msg())
        << 1
    }

    first := db.exec(
        "INSERT INTO xu_people (id, name, active) VALUES (?, ?, ?)",
        1,
        "Ada",
        true
    ) or (err) {
        println(err.msg())
        << 1
    }
    println("inserted=" + str(first.rows_affected()))

    mut insert_person := db.prepare(
        "INSERT INTO xu_people (id, name, active) VALUES (?, ?, ?)"
    ) or (err) {
        println(err.msg())
        << 1
    }
    insert_person.exec(2, "Lin", true) or (err) {
        println(err.msg())
        << 1
    }
    insert_person.close()

    mut tx := db.begin() or (err) {
        println(err.msg())
        << 1
    }
    tx.exec(
        "INSERT INTO xu_people (id, name, active) VALUES (?, ?, ?)",
        3,
        "Grace",
        true
    ) or (err) {
        println(err.msg())
        << 1
    }
    tx.commit() or (err) {
        println(err.msg())
        << 1
    }

    mut rows := db.query("SELECT id, name FROM xu_people ORDER BY id") or (err) {
        println(err.msg())
        << 1
    }
    while rows.next() {
        values := rows.take() or (err) {
            println(err.msg())
            << 1
        }
        id := values[0] as int
        name := values[1] as string
        println(str(id) + ": " + name)
    }
    if rows.err().len > 0 {
        println(rows.err())
        << 1
    }
    rows.close()

    row := db.query_row("SELECT name FROM xu_people WHERE id = ?", 2)
    values := row.take() or (err) {
        println(err.msg())
        << 1
    }
    println("selected=" + (values[0] as string))

    db.close() or (err) {
        println(err.msg())
        << 1
    }
    << 0
}

// PostgreSQL：XU_POSTGRES_DSN 可覆盖本地开发默认值。
// 驱动是普通 XU 包，通过 database/sql 接口使用，不依赖编译器内建。

fn main() int {
    mut dsn := os.getenv("XU_POSTGRES_DSN")
    if dsn.len == 0 {
        dsn = "postgres://postgres:postgres@127.0.0.1:5432/xu_demo?sslmode=disable"
    }

    mut db := sql.open(postgres.driver(), dsn)
    db.ping() or (err) {
        println("PostgreSQL ping failed")
        println(err.msg())
        << 1
    }

    db.exec("DROP TABLE IF EXISTS xu_people") or (err) {
        println(err.msg())
        << 1
    }
    db.exec("CREATE TABLE xu_people (id INTEGER PRIMARY KEY, name VARCHAR(100) NOT NULL, active BOOLEAN NOT NULL)") or (err) {
        println(err.msg())
        << 1
    }

    first := db.exec(
        "INSERT INTO xu_people (id, name, active) VALUES ($$1, $$2, $$3)",
        1,
        "Ada",
        true
    ) or (err) {
        println(err.msg())
        << 1
    }
    println("inserted=" + str(first.rows_affected()))

    mut insert_person := db.prepare(
        "INSERT INTO xu_people (id, name, active) VALUES ($$1, $$2, $$3)"
    )
    insert_person.exec(2, "Lin", true) or (err) {
        println(err.msg())
        << 1
    }
    insert_person.close()

    mut tx := db.begin() or (err) {
        println(err.msg())
        << 1
    }
    tx.exec(
        "INSERT INTO xu_people (id, name, active) VALUES ($$1, $$2, $$3)",
        3,
        "Grace",
        true
    ) or (err) {
        println(err.msg())
        << 1
    }
    tx.commit() or (err) {
        println(err.msg())
        << 1
    }

    mut rows := db.query("SELECT id, name FROM xu_people ORDER BY id") or (err) {
        println(err.msg())
        << 1
    }
    while rows.next() {
        values := rows.take() or (err) {
            println(err.msg())
            << 1
        }
        id := values[0] as int
        name := values[1] as string
        println(str(id) + ": " + name)
    }
    if rows.err().len > 0 {
        println(rows.err())
        << 1
    }
    rows.close()

    mut qrows := db.query("SELECT name FROM xu_people WHERE id = $$1", 2) or (err) {
        println(err.msg())
        << 1
    }
    if qrows.next() {
        vals := qrows.take() or (err) {
            println(err.msg())
            << 1
        }
        println("selected=" + (vals[0] as string))
    }
    qrows.close()

    db.close() or (err) {
        println(err.msg())
        << 1
    }
    << 0
}
