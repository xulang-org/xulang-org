Грамматика EBNF

Формальное описание синтаксиса Xu в нотации Бэкуса — Наура (EBNF):

SourceFile   = { CompileTag } { TopLevelDecl } ;
CompileTag   = "$" ( "freestanding" | "macos" | "linux" | "windows" | Ident ) ;
TopLevelDecl = [ "@main" ] [ Attribute ] ( VarDecl | ConstDecl | FuncDecl | StructDecl | EnumDecl ) ;
Attribute    = "#[" Ident [ "(" AttrArg ")" ] "]" | "@repr" "(" StringLit [ "," ReprArg ] ")" ;

FuncDecl     = "fn" [ Receiver ] Ident Signature Block ;
Receiver     = "(" Ident [ [ "$" ] "&" ] Type ")" ;
Signature    = "(" [ ParameterList ] ")" [ Type ] ;

Block        = "{" { Statement } "}" ;
Statement    = VarDecl | Assignment | ReturnStmt | ThrowStmt | BreakStmt | IfStmt | ForStmt | ExprStmt | UnsafeBlock | AsmStmt ;

ReturnStmt   = "<<" [ Expression ] ;
ThrowStmt    = ">>" Expression ;
BreakStmt    = "<-" ;
UnsafeBlock  = "unsafe" Block ;
AsmStmt      = "asm!" "(" StringLit { "," AsmOperand } ")" ;
RawPointer   = "*raw" Type ;
