07 · Массивы и срезы

Динамические массивы поддерживают добавление элементов через push. Срезы строк s[a..b] работают эффективно без копирования буферов.

@main
#fmt

fn main() int {
    $nums := []int{}
    nums.push(10)
    nums.push(20)
    nums.push(30)

    fmt.println("Размер массива: " + nums.len.to_string())
    for n in nums {
        fmt.println(n)
    }

    text := "Hello World"
    slice := text[0..5]
    fmt.println(slice)
    << 0
}
