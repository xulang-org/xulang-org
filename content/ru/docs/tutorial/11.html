11 · Низкоуровневый HTTP-сервер

Создание HTTP-сервера с собственным циклом listen / accept и сопоставлением путей.

@main
#net/http
#fmt

fn main() int {
    server := http.listen("127.0.0.1:8080") or (err) {
        fmt.println("Ошибка запуска: " + err.msg)
        << 1
    }
    fmt.println("Сервер слушает на http://127.0.0.1:8080")

    for {
        conn := server.accept() or (err) { <- }
        xu handle(conn)
    }
    << 0
}

fn handle(conn http.Conn) {
    conn.write_response(200, "OK", "Hello from raw HTTP in Xu!")
    conn.close()
}
