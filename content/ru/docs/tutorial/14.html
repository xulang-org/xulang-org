14 · TCP эхо-сервер

Сетевой TCP сервер: прослушивание порта, принятие подключений и обработка в корутинах.

@main
#net
#fmt

fn handle_client(conn net.Conn) {
    $buf := [1024]byte{}
    for {
        n := conn.read($&buf) or (err) { <- }
        if n == 0 { <- }
        conn.write(buf[0..n]) or (err) { <- }
    }
    conn.close()
}

fn main() int {
    ln := net.listen("tcp", "127.0.0.1:9000") or (err) { << 1 }
    fmt.println("TCP Echo слушает на :9000")
    for {
        conn := ln.accept() or (err) { <- }
        xu handle_client(conn)
    }
    << 0
}
