13 · Маршрутизация HTTP Context и Mux

Использование встроенного маршрутизатора ServeMux и объекта запроса http.Context:

@main
#net/http
#fmt

fn handle_home(ctx $&http.Context) {
    ctx.text(200, "Добро пожаловать на главную страницу!")
}

fn handle_api(ctx $&http.Context) {
    ctx.json(200, "{\"status\": \"ok\"}")
}

fn main() int {
    $mux := http.new_mux()
    mux.handle("/", handle_home)
    mux.handle("/api", handle_api)

    http.serve("127.0.0.1:8080", &mux) >>> (err) { << 1 }
    << 0
}
