10 · Корутины и каналы

Ключевое слово xu запускает функцию в виде легковесной асинхронной корутины. Передача данных и синхронизация выполняются через типизированные каналы chan.

@main
#fmt

fn producer(ch chan int) {
    for i in 1..=3 {
        ch <- i * 10
    }
}

fn main() int {
    ch := chan int{cap: 3}
    xu producer(ch)

    fmt.println(<-ch)
    fmt.println(<-ch)
    fmt.println(<-ch)
    << 0
}
