05 · Обработка ошибок

Xu не использует сложные конструкции вроде Result/unwrap или вездесущие проверки if err != nil. Ошибки выбрасываются оператором >> и обязательно обрабатываются в месте вызова блоком or (err) { ... }.

@main
#fmt
#errors

fn divide(a int, b int) int {
    if b == 0 {
        >> errors.new("деление на ноль")
    }
    << a / b
}

fn main() int {
    res := divide(10, 0) or (err) {
        fmt.println("Ошибка перехвачена: " + err.msg)
        << 1
    }
    fmt.println(res)
    << 0
}
