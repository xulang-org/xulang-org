12 · HTTP файловый сервер

Раздача статических файлов из каталога в одну строку:

@main
#net/http
#fmt

fn main() int {
    fmt.println("Раздача файлов на http://127.0.0.1:8080")
    http.listen_and_serve("127.0.0.1:8080", http.file_server("./public")) >>> (err) {
        fmt.println("Ошибка: " + err.msg)
        << 1
    }
    << 0
}
