15 · UDP эхо-сервер

Сетевой протокол UDP без установления соединения: приём дейтаграммы и отправка ответа отправителю.

@main
#net
#fmt

fn main() int {
    socket := net.listen_packet("udp", "127.0.0.1:9001") or (err) { << 1 }
    fmt.println("UDP Echo слушает на :9001")
    $buf := [2048]byte{}
    for {
        res := socket.read_from($&buf) or (err) { <- }
        socket.write_to(buf[0..res.bytes], res.addr) or (err) { <- }
    }
    << 0
}
