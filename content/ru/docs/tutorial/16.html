16 · Настольный UI и системный WebView

Модуль #ui предоставляет оконную подсистему, меню и трей, а #ui/webview обеспечивает интеграцию современного браузерного интерфейса.

@main
#ui
#ui/webview

fn main() int {
    app := ui.new_app("org.example.myapp")
    win := app.create_window(ui.WindowOptions{
        title: "Приложение Xu",
        width: 1024,
        height: 768
    })

    wv := webview.create(win, webview.Options{
        url: "https://xulang.org"
    })

    app.run()
    << 0
}
