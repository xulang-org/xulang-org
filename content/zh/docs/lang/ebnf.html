EBNF

规划语法浓缩（用于对照，不是 parser 逐字表）：

File         = "@" DottedPath { CompileTag } { Import } { Decl } .
Import       = [ Ident ] "#" Path [ "as" Ident ] .
ReturnStmt   = "<<" [ Expr ] .
ThrowStmt    = ">>" Expr .
ExitStmt     = "<<<" Expr .
BreakStmt    = "<-" .
ContinueStmt = "->" .
RangeOp      = ":" . // [start, end)
XuStmt       = "xu" Ident "(" [ Args ] ")" .
Expr         = … | Expr "or" "(" Ident ")" Block | Expr "or" ">>" .
