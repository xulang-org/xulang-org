fn paint(context &draw.Context,frame $&draw.Frame,input &ui.TextInput) int {
    state := input.snapshot() or >>
    value := state.display_value()
    println("Text revision: " + str(state.revision) + " / display bytes: " + str(value.len))
    options := draw.text_options(28.0)
    $text := context.layout_text(&value,&options) or >>
    text.bind_input(input,state.revision,44.0,180.0) or >>
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,28.0,28.0,14.0,demo.accent(),"04 / NATIVE TEXT INPUT") or >>
    demo.text(context,frame,28.0,60.0,28.0,demo.ink(),"Type in your own language.") or >>
    demo.text(context,frame,28.0,112.0,15.0,demo.muted(),"Use an IME, arrows, Shift+arrows and Backspace. Click to move the caret.") or >>
    frame.fill_rect(28.0,160.0,790.0,86.0,draw.rgb(26,37,50)) or >>
    anchor := text.caret(state.display_anchor(),false) or >>
    caret := text.caret(state.display_focus(),false) or >>
    if state.display_anchor() != state.display_focus() && anchor.y == caret.y {
        frame.fill_rect(44.0 + math.min(anchor.x,caret.x),180.0 + caret.y,math.abs(caret.x - anchor.x),caret.height,draw.rgba(78,205,196,80)) or >>
    }
    frame.text_layout(&text,44.0,180.0,demo.ink()) or >>
    frame.fill_rect(44.0 + caret.x,180.0 + caret.y,1.5,caret.height,demo.accent()) or >>
    if state.composing {
        demo.text(context,frame,28.0,280.0,17.0,demo.accent(),"Composing: " + state.preedit) or >>

    }

fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    seed := "你好，Xu 👩‍💻 · café"
    caret := ui.TextInputRect{x: 44.0,y: 180.0,width: 2.0,height: 36.0}
    options := ui.text_input_options(&seed,caret)
    $input := ui.start_text_input(identity.canvas_id,identity.generation,&options) or >>
    $frame := context.new_frame(4096)
    paint(&context,$&frame,&input) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        $dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind >= ui.EV_TEXT_COMMIT && event.kind <= ui.EV_TEXT_SURROUNDING_DELETE {
            edit := ui.text_edit_of(&event) or >>
            if edit.input.id == input.id {
                if event.kind == ui.EV_TEXT_INPUT_ENDED || event.kind == ui.EV_TEXT_INPUT_FAILED { <- }
                dirty = true
            }
        }
        if ui.is_input_event(event.kind) {
            pointer := ui.input_of(&event) or >>
            if pointer.generation != identity.generation { -> }
            if pointer.kind == ui.EV_KEY_DOWN && pointer.key == ui.KEY_ESCAPE { demo.quit() or >> }
            bounds := layout.Rect{x: 28.0,y: 160.0,width: 790.0,height: 86.0}
            if pointer.kind == ui.EV_POINTER_DOWN && pointer.button == ui.BUTTON_LEFT && bounds.contains(pointer.x,pointer.y) {
                state := input.snapshot() or >>
                // The native editor owns preedit; do not overwrite an active composition.
                if !state.composing {
                    hit := input.hit_test(state.revision,pointer.x,pointer.y) or >>
                    index := hit.byte_index()
                    _ := input.update(state.revision,&state.value,index,index,&caret) or >>
                    dirty = true
                }
            }
        }
        if dirty { paint(&context,$&frame,&input) or >> }
    }
    frame.dispose()
    input.stop() or (err) { << 0 }
    << 0
}

fn (m $&Model) update(screen $&widgets.Screen,action widgets.Action) int {
    if action.kind == widgets.ACTION_KEY && int(action.value) == ui.KEY_ESCAPE { screen.close() or >> }
    if action.kind == widgets.ACTION_ACTIVATE {
        for i in 0:3 { if action.id == m.nav[i] { m.page = i } }
        if action.id == m.settings[0] { m.theme = (m.theme + 1) % 3
            current := screen.styles()
            $next := widgets.preset_styles(m.theme) or >>
            next.text_scale = current.text_scale
            next.density = current.density
            screen.set_styles(&next) or >> }
        if action.id == m.settings[1] || action.id == m.settings[2] {
            $styles := screen.styles()
            if action.id == m.settings[1] { if styles.text_scale == 1.0 { styles.text_scale = 1.4 }
                else { styles.text_scale = 1.0 } }
            else { if styles.density == 1.0 { styles.density = 0.8 }
                else { styles.density = 1.0 } }
            screen.set_styles(&styles) or >>
        }
        if action.id == m.table { m.fill_table(screen) or >> }
        if action.id == m.overlay_buttons[0] { screen.open(m.popover) or >> }
        if action.id == m.overlay_buttons[1] { screen.open(m.modal) or >> }
        if action.id == m.overlay_buttons[2] { _ := screen.notify("All components share the same theme.") or >> }
        if action.id == m.modal_actions[0] || action.id == m.modal_actions[1] { screen.close_overlay(m.modal) or >>
            if action.id == m.modal_actions[0] { _ := screen.notify("Action completed.") or >> } }
    }
    if action.kind == widgets.ACTION_CHANGE {
        if action.id == m.pager { m.fill_table(screen) or >> }
        if action.id == m.combo || action.id == m.multi {
            value := screen.text_value(m.combo) or >>
            count := (screen.values(m.multi) or >>).len
            screen.set_text(m.value_label,"Value: " + value + "    Selected tags: " + str(count)) or >>
        }
        screen.set_text(m.note,"Value changed · Tab / Shift+Tab · Escape") or >>
    }
    << 0
}
