fn paint(context &draw.Context,frame &mut draw.Frame,input &ui.TextInput) int {
    state := input.snapshot() or >>
    value := state.display_value()
    println("Text revision: " + str(state.revision) + " / display bytes: " + str(value.len))
    options := draw.text_options(28.0)
    mut text := context.layout_text(&value,&options) or >>
    text.bind_input(input,state.revision,44.0,180.0) or >>
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,28.0,28.0,14.0,demo.accent(),"04 / NATIVE TEXT INPUT") or >>
    demo.text(context,frame,28.0,60.0,28.0,demo.ink(),"Type in your own language.") or >>
    demo.text(context,frame,28.0,112.0,15.0,demo.muted(),"Use an IME, arrows, Shift+arrows and Backspace. Click to move the caret.") or >>
    frame.fill_rect(28.0,160.0,790.0,86.0,draw.rgb(26,37,50)) or >>
    anchor := text.caret(state.display_anchor(),false) or >>
    caret := text.caret(state.display_focus(),false) or >>
    if state.display_anchor() != state.display_focus() && anchor.y == caret.y {
        frame.fill_rect(44.0 + math.min(anchor.x,caret.x),180.0 + caret.y,math.abs(caret.x - anchor.x),caret.height,draw.rgba(78,205,196,80)) or >>
    }
    frame.text_layout(&text,44.0,180.0,demo.ink()) or >>
    frame.fill_rect(44.0 + caret.x,180.0 + caret.y,1.5,caret.height,demo.accent()) or >>
    if state.composing {
        demo.text(context,frame,28.0,280.0,17.0,demo.accent(),"Composing: " + state.preedit) or >>
    } else {
        demo.text(context,frame,28.0,280.0,17.0,demo.muted(),"Committed text / no active preedit") or >>
    }
    demo.text(context,frame,28.0,324.0,14.0,demo.muted(),"Revision " + str(state.revision) + " / anchor byte " + str(state.anchor) + " / focus byte " + str(state.focus)) or >>
    demo.text(context,frame,28.0,360.0,14.0,demo.muted(),"Selection uses UTF-8 byte offsets. Escape ends the session and closes the window.") or >>
    frame.present() or >>
    text.dispose() or >>
    << 0
}
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    seed := "你好，Xu 👩‍💻 · café"
    caret := ui.TextInputRect{x: 44.0,y: 180.0,width: 2.0,height: 36.0}
    options := ui.text_input_options(&seed,caret)
    mut input := ui.start_text_input(identity.canvas_id,identity.generation,&options) or >>
    mut frame := context.new_frame(4096)
    paint(&context,mut &frame,&input) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT || event.kind == ui.EV_CANVAS_CLOSED { <- }
        if event.window_id != identity.window_id { -> }
        mut dirty := event.kind == ui.EV_CANVAS_METRICS
        if event.kind >= ui.EV_TEXT_COMMIT && event.kind <= ui.EV_TEXT_SURROUNDING_DELETE {
            edit := ui.text_edit_of(&event) or >>
            if edit.input.id == input.id {
                if event.kind == ui.EV_TEXT_INPUT_ENDED || event.kind == ui.EV_TEXT_INPUT_FAILED { <- }
                dirty = true
            }
        }
        if ui.is_input_event(event.kind) {
            pointer := ui.input_of(&event) or >>
            if pointer.generation != identity.generation { -> }
            if pointer.kind == ui.EV_KEY_DOWN && pointer.key == ui.KEY_ESCAPE { demo.quit() or >> }
            bounds := layout.Rect{x: 28.0,y: 160.0,width: 790.0,height: 86.0}
            if pointer.kind == ui.EV_POINTER_DOWN && pointer.button == ui.BUTTON_LEFT && bounds.contains(pointer.x,pointer.y) {
                state := input.snapshot() or >>
                // The native editor owns preedit; do not overwrite an active composition.
                if !state.composing {
                    hit := input.hit_test(state.revision,pointer.x,pointer.y) or >>
                    index := hit.byte_index()
                    _ := input.update(state.revision,&state.value,index,index,&caret) or >>
                    dirty = true
                }
            }
        }
        if dirty { paint(&context,mut &frame,&input) or >> }
    }
    frame.dispose()
    input.stop() or (err) { << 0 }
    << 0
}
fn worker(ech chan ui.Event) { scene(ech) or (err) { demo.fail(err.msg())
    << } }
fn launch() int {
    mut app := demo.app("Xu UI / Native Text Input",860,430) or >>
    xu worker(app.event_ch)
    << app.run() or >>
}
fn main() int { << launch() or (err) { eprintln(err.msg())
    << 1 } }

struct Model {
    button widgets.Button
    checkbox widgets.Checkbox
    progress widgets.Progress
    scroll widgets.Scroll
    button_label draw.TextLayout
    checkbox_label draw.TextLayout
    theme widgets.Theme
    context draw.Context
    count int
    message string
    source string
    images []draw.Image
    header_id int
    button_id int
    check_id int
    progress_id int
    scroll_id int
    row_ids []int
}
fn (model &mut Model) button_input(kind int,input &ui.InputEvent) widgets.Action {
    button := mut &model.button
    << button.route(kind,input) or >>
}
fn (model &mut Model) checkbox_input(kind int,input &ui.InputEvent) widgets.Action {
    checkbox := mut &model.checkbox
    << checkbox.route(kind,input) or >>
}
fn (model &mut Model) scroll_input(kind int,input &ui.InputEvent) widgets.Action {
    scroll := mut &model.scroll
    << scroll.route(kind,input) or >>
}
fn (model &mut Model) input(id int,kind int,input &ui.InputEvent) widgets.Action {
    if id == model.button_id { << model.button_input(kind,input) or >> }
    if id == model.check_id { << model.checkbox_input(kind,input) or >> }
    if id == model.scroll_id { << model.scroll_input(kind,input) or >> }
    << widgets.Action{kind: widgets.ACTION_NONE,id: id,needs_paint: false,request_focus: false,consumed: false,value: 0.0}
}
fn (model &mut Model) place_button(bounds &layout.Rect) int {
    button := mut &model.button
    << button.set_bounds(bounds) or >>
}
fn (model &mut Model) place_checkbox(bounds &layout.Rect) int {
    checkbox := mut &model.checkbox
    << checkbox.set_bounds(bounds) or >>
}
fn (model &mut Model) place_progress(bounds &layout.Rect) int {
    progress := mut &model.progress
    << progress.set_bounds(bounds) or >>
}
fn (model &mut Model) paint(id int,bounds &layout.Rect,frame &mut draw.Frame) bool {
    model.draw_node(id,bounds,frame) or (err) { eprintln(err.msg())
        << false }
    << true
}
fn (model &mut Model) draw_node(id int,bounds &layout.Rect,frame &mut draw.Frame) int {
    if id == model.header_id {
        demo.text(&model.context,frame,0.0,0.0,14.0,demo.accent(),"07 / IMAGE WORKBENCH") or >>
        demo.text(&model.context,frame,0.0,28.0,24.0,demo.ink(),"Image workbench") or >>
        demo.text(&model.context,frame,0.0,58.0,13.0,demo.muted(),model.message + "") or >>
    } else if id == model.button_id {
        model.place_button(bounds) or >>
        model.button.paint(frame,&model.button_label,&model.theme) or >>
    } else if id == model.check_id {
        model.place_checkbox(bounds) or >>
        model.checkbox.paint(frame,&model.checkbox_label,&model.theme) or >>
    } else if id == model.progress_id {
        model.place_progress(bounds) or >>
        model.progress.paint(frame,&model.theme) or >>
    } else if id == model.scroll_id {
        scroll := mut &model.scroll
        scroll.set_bounds(bounds) or >>
        frame.fill_rect(0.0,0.0,bounds.width,bounds.height,draw.rgb(26,37,50)) or >>
    } else if model.row_ids.len > 0 && id == model.row_ids[0] {
        if model.images.len > 0 {
            image := model.images[0]
            scale := math.min(bounds.width / f64(image.width),bounds.height / f64(image.height))
            frame.image(&image,0.0,0.0,f64(image.width) * scale,f64(image.height) * scale) or >>
        } else { demo.text(&model.context,frame,12.0,20.0,17.0,demo.muted(),"The selected image appears here.") or >> }
    }
    << 0
}
fn (model &mut Model) progress_value(value f64) int {
    progress := mut &model.progress
    << progress.set_value(value) or >>
}
fn (model &mut Model) retire_image(index int) int {
    mut image := model.images[index]
    image.dispose() or >>
    model.images[index] = image
    << 0
}
fn (model &mut Model) replace_image(image draw.Image) int {
    for i in 0:model.images.len { model.retire_image(i) or >> }
    model.images = []draw.Image{}
    model.images.push(image)
    << 0
}
fn preview_paint(window &ui.Window,path &string) int {
    context := draw.window_context(window) or >>
    mut image := context.load_image(path) or >>
    mut frame := context.new_frame(256)
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    scale := math.min(f64(window.width - 40) / f64(image.width),f64(window.height - 40) / f64(image.height))
    frame.image(&image,20.0,20.0,f64(image.width) * scale,f64(image.height) * scale) or >>
    frame.present() or >>
    frame.dispose()
    image.dispose() or >>
    << 0
}

fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.EV_EXIT { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    host := widgets.Host{context: context,identity: identity}
    mut column := layout.box_style(layout.BOX_COLUMN)
    column.padding = layout.insets(28.0) or >>
    column.gap = 14.0
    column.cross_alignment = layout.CROSS_STRETCH
    mut view := host.view(&column,demo.background()) or >>
    leaf := layout.box_style(layout.BOX_LEAF)
    header := view.add(view.root(),&leaf,&layout.Size{width: 400.0,height: 88.0}) or >>
    button_id := view.add(view.root(),&leaf,&layout.Size{width: 280.0,height: 44.0}) or >>
    check_id := view.add(view.root(),&leaf,&layout.Size{width: 280.0,height: 40.0}) or >>
    progress_id := view.add(view.root(),&leaf,&layout.Size{width: 280.0,height: 12.0}) or >>
    mut viewport := layout.box_style(layout.BOX_COLUMN)
    viewport.height = 260.0
    viewport.clip = true
    viewport.cross_alignment = layout.CROSS_STRETCH
    viewport.gap = 6.0
    scroll_id := view.add(view.root(),&viewport,&layout.Size{width: 400.0,height: 260.0}) or >>
    mut rows := []int{}
    for _ in 0:1 {
        row := view.add(scroll_id,&leaf,&layout.Size{width: 400.0,height: 250.0}) or >>
        rows.push(row)
        view.set_boundary(row,true) or >>
    }
    for id in [header,button_id,check_id,progress_id,scroll_id] { view.set_boundary(id,true) or >> }
    for id in [button_id,check_id,scroll_id] { view.set_input(id,true,true) or >> }
    empty := layout.Rect{x: 0.0,y: 0.0,width: 0.0,height: 0.0}
    options := draw.text_options(15.0)
    first := "Open a PNG or JPEG"
    second := "Show a separate preview window"
    mut model := Model{button: widgets.button(&identity,button_id,&empty) or >>,
        checkbox: widgets.checkbox(&identity,check_id,&empty,true) or >>,
        progress: widgets.progress(&empty,0.0) or >>,
        scroll: widgets.scroll(&identity,scroll_id,&empty,widgets.AXIS_VERTICAL,250.0) or >>,
        button_label: context.layout_text(&first,&options) or >>,
        checkbox_label: context.layout_text(&second,&options) or >>,
        theme: widgets.default_theme(),context: context,count: 0,message: "Open a local image to begin.",source: "",images: []draw.Image{},header_id: header,button_id: button_id,check_id: check_id,progress_id: progress_id,scroll_id: scroll_id,row_ids: rows}
    mut width := 900.0
    mut height := 700.0
    mut native := demo.windows(ech)
    mut preview := ui.Window{id: 0,generation: 0,title: "Preview",width: 720,height: 500,content_kind: ui.CONTENT_DRAW,content_backend_id: 0,closed: 1}
    mut pending := ui.Request{id: 0,window_id: 0,canvas_id: 0,generation: 0,kind: 0}
    while true {
        constraints := layout.tight(width,height) or >>
        submitted := widgets.present(mut &view,&constraints,mut &model) or >>
        if submitted { model.acknowledge() }
        event := demo.next(ech)
        if event.kind == ui.EV_EXIT { <- }
        if preview.id > 0 && event.window_id == preview.id {
            if event.kind == ui.EV_CLOSED || event.kind == ui.EV_WINDOW_FAILED { preview.closed = 1 }
            if event.kind == ui.EV_CANVAS_READY && model.source.len > 0 {
                preview.content_command("input_enabled\t1") or >>
                preview.content_command("pan_enabled\t0") or >>
                preview_paint(&preview,&model.source) or >> }
            if event.kind == ui.EV_CANVAS_METRICS {
                metrics := ui.metrics_of(&event) or >>
                if metrics.width > 40.0 && metrics.height > 40.0 { preview.width = int(metrics.width)
                    preview.height = int(metrics.height)
                    if model.source.len > 0 { preview_paint(&preview,&model.source) or >> } }
            }
            ->
        }
        if event.window_id != identity.window_id { -> }
        if event.kind == ui.EV_REQUEST {
            result := ui.request_of(&event) or >>
            if result.matches(&pending) {
                pending.id = 0
                model.message = "Selection cancelled or failed."
                if result.status == ui.REQUEST_COMPLETED {
                    paths := ui.file_paths(&result) or >>
                    if paths.len > 0 {
                        path := paths[0] + ""
                        image := context.load_image(&path) or (err) { model.message = err.msg()
                            view.mark_paint(header) or >>
                            -> }
                        model.replace_image(image) or >>
                        model.source = path + ""
                        model.message = "Loaded " + str(image.width) + " x " + str(image.height) + " pixels"
                        println(model.message + "")
                        model.progress_value(1.0) or >>
                        view.mark_paint(model.row_ids[0]) or >>
                        if model.checkbox.value() {
                            if preview.closed != 0 { preview = native.open_content_window(ui.WindowOpts{title: "Xu UI / Image Preview",width: 720,height: 500},ui.CONTENT_DRAW) or >> }
                            else { preview_paint(&preview,&path) or >>
                                preview.focus() or >> }
                        }
                    }
                }
                view.mark_paint(header) or >>
                view.mark_paint(progress_id) or >>
            }
        }
        if event.kind == ui.EV_CANVAS_METRICS {
            metrics := ui.metrics_of(&event) or >>
            if metrics.width > 0.0 && metrics.height > 0.0 { width = metrics.width
                height = metrics.height }
        }
        actions := widgets.dispatch_event(mut &view,&event,mut &model) or >>
        for action in actions {
            if action.kind == widgets.ACTION_ACTIVATE && pending.id == 0 {
                mut options := ui.file_dialog_options()
                options.title = "Open an image"
                options.filters = [ui.FileFilter{label: "Images",extensions: ["png","jpg","jpeg"]}]
                pending = ui.open_files_async(&identity,&options) or >>
                model.message = "Choose a PNG or JPEG in the native dialog."
                model.progress_value(0.2) or >>
                view.mark_paint(header) or >>
                view.mark_paint(progress_id) or >>
            }
            if action.kind == widgets.ACTION_SCROLL { view.set_scroll(scroll_id,0.0,action.value) or >> }
        }
        if event.kind == ui.EV_CANVAS_CLOSED { demo.quit() or >>
            <- }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.kind == ui.EV_KEY_DOWN && input.key == ui.KEY_ESCAPE { demo.quit() or >> }
        }
        stats := view.stats()
        if stats.recorded_boundaries > 0 || stats.reused_boundaries > 0 {
            println("View: " + str(stats.recorded_boundaries) + " recorded / " + str(stats.reused_boundaries) + " reused boundaries")
        }
    }
    if pending.id > 0 { _ := pending.cancel() }
    view.dispose()
    // Native close has already revoked this canvas's images and text layouts.
    if demo.alive(&identity) {
        for i in 0:model.images.len { model.retire_image(i) or >> }
        label := mut &model.button_label
        label.dispose() or >>
        check_label := mut &model.checkbox_label
        check_label.dispose() or >>
    }
    << 0
}
