提供文本响应

使用 net/http 在请求处理函数中返回 HTTP 文本响应。

此原生示例监听本机回环地址；按 Ctrl-C 停止。在另一终端运行 curl http://127.0.0.1:8080/。

将完整程序保存为 main.xu，在本机运行 xu run main.xu。

}

预期响应正文：hello，末尾有换行。

服务端在连接接收、缓冲区和超时限额内并发派发完整请求。处理器可以重叠执行，共享可变状态须同步，并显式传递请求所有权。使用 server_options() 获取默认值，通过 serve_with_options 配置限额；响应短写会重试，空闲或不完整请求不会无限占用处理器。

// 写法 C — 请求上下文（Go Handler / HandlerFunc）。
//
// Go:
//   func hello(w http.ResponseWriter, r *http.Request) {
//       fmt.Fprint(w, "hello")
//   }
//   http.ListenAndServe(":8080", http.HandlerFunc(hello))
//
// Xu: w 和 r 收成一个 Context。

fn app(c http.Context) {
    p := c.path()
    if p == "/" || p == "/index.html" {
        page := "<!doctype html><html><head><meta charset=\"utf-8\"><title>Xu</title></head>" +
            "<body><h1>Context</h1>" +
            "<p><a href=\"/hello\">/hello</a> · <a href=\"/health\">/health</a></p>" +
            "<p>POST /echo</p></body></html>"
        c.html(200, page) or (err) { << }
        <<
    }
    if p == "/hello" {
        c.text(200, "hello, xu\n") or (err) { << }
        <<
    }
    if p == "/health" {
        c.text(200, "ok\n") or (err) { << }
        <<
    }
    if p == "/echo" {
        m := c.method()
        if m != "POST" && m != "PUT" {
            c.text(405, "method not allowed\n") or (err) { << }
            <<
        }
        b := c.body()
        c.text(200, b) or (err) { << }
        <<
    }
    miss := "not found: " + p + "\n"
    c.text(404, miss) or (err) { << }
}

fn main() int {
    println("http://127.0.0.1:18081")
    http.listen_and_serve_handler("127.0.0.1:18081", app) or (err) {

        << 1
    }
    << 0

// Xu Design System & VI Specification 官方本地 HTTP 服务
// 运行指令：
//   ./xu_design_server.xin [--addr 127.0.0.1:8088] [--root <www-dir>]
//   位置参数仍兼容旧形式：<port> <www-dir>。

fn main() int {
    mut addr := "127.0.0.1:8088"
    mut root := "examples/16_ui_webview/xu_design_system/www"

    av := args()
    mut i := 1
    while i < av.len {
        arg := av[i] + ""
        if arg == "--help" || arg == "-h" {
            println("Xu Design System HTTP server")
            println("usage: xu_design_server [--addr HOST:PORT] [--root DIR]")
            println("legacy: xu_design_server PORT DIR")
            << 0
        }
        if arg == "--addr" {
            if i + 1 >= av.len {
                println("missing value for --addr")
                << 2
            }
            addr = av[i + 1] + ""
            i = i + 2
            ->
        }
        if arg == "--root" {
            if i + 1 >= av.len {
                println("missing value for --root")
                << 2
            }
            root = av[i + 1] + ""
            i = i + 2
            ->
        }
        // Preserve the original positional interface: port, then root.
        if i == 1 {
            addr = "127.0.0.1:" + arg
        } else if i == 2 {
            root = arg
        } else {
            println("unknown argument: " + arg)
            << 2
        }
        i = i + 1
    }

    println("==================================================================")
    println("    __  __  _   _  ")
    println("    \\ \\/ / | | | |   ⚡ Xu Official HTTP Server (Design System)")
    println("     \\  /  | | | |   🔥 Rock-Solid Ownership · Supersonic AOT")
    println("     /  \\  | |_| |   🌐 Serving: " + root)
    println("    /_/\\_\\  \\___/    🔗 Local URL: http://" + addr + "/")
    println("==================================================================")
    println("Press Ctrl+C to stop server.\n")

    http.listen_and_serve(addr, root) or (err) {
        println("Server error: " + err.msg())
        << 1
    }
    << 0
}

// 自己 listen / accept / 按路径处理（Go HandleFunc 那种）。
//   GET  /        HTML
//   GET  /hello   text
//   GET  /health  ok
//   POST /echo    回显 body
// 静态目录见 12file_server；Handler 上下文见 13http_context。

fn path_only(p string) string {
    mut i := 0
    while i < p.len {
        if (p[i] & 255) == 63 {
            << p[0:i]
        }
        i = i + 1
    }
    << p + ""
}

fn write_http(fd int, status int, ctype string, body string) int {
    mut reason := "OK"
    if status == 404 {
        reason = "Not Found"
    }
    if status == 400 {
        reason = "Bad Request"
    }
    if status == 405 {
        reason = "Method Not Allowed"
    }
    mut out := "HTTP/1.1 " + str(status) + " " + reason + "\r\n"
    out = out + "Content-Type: " + ctype + "\r\n"
    out = out + "Content-Length: " + str(body.len) + "\r\n"
    out = out + "Connection: close\r\n"
    out = out + "\r\n"
    out = out + body
    << net.conn_write(fd, &out) or (err) { >> err }
}

fn index_page() string {
    << "<!doctype html><html><head><meta charset=\"utf-8\"><title>Xu HTTP</title>" +
        "<style>body{font-family:ui-sans-serif,system-ui,sans-serif;max-width:40rem;" +
        "margin:3rem auto;padding:0 1.5rem;background:#0f1419;color:#e7ecf1}" +
        "h1{font-weight:650;letter-spacing:-.03em}a{color:#7dd3fc}" +
        "code{background:#1b2430;padding:.15rem .4rem;border-radius:4px}" +
        "ul{line-height:1.8}</style></head><body>" +
        "<h1>Xu HTTP</h1>" +
        "<p>自己 <code>listen</code> / <code>accept</code> / 按 path 分支。</p>" +
        "<ul>" +
        "<li><a href=\"/hello\">/hello</a></li>" +
        "<li><a href=\"/health\">/health</a></li>" +
        "<li>POST <code>/echo</code></li>" +
        "</ul></body></html>"
}

fn handle(fd int) {
    mut req := http.new_request()
    req.read(fd) or (err) {
        msg := err.msg()
        write_http(fd, 400, "text/plain; charset=utf-8", msg) or (e2) { << }
        net.conn_close(fd) or (e3) { << }
        <<
    }
    method := req.method()
    raw_path := req.path()
    path := path_only(raw_path)
    if path == "/" || path == "/index.html" {
        page := index_page()
        write_http(fd, 200, "text/html; charset=utf-8", page) or (err) { << }
        net.conn_close(fd) or (err) { << }
        <<
    }
    if path == "/hello" {
        body := "hello, xu\n"
        write_http(fd, 200, "text/plain; charset=utf-8", body) or (err) { << }
        net.conn_close(fd) or (err) { << }
        <<
    }
    if path == "/health" {
        body := "ok\n"
        write_http(fd, 200, "text/plain; charset=utf-8", body) or (err) { << }
        net.conn_close(fd) or (err) { << }
        <<
    }
    if path == "/echo" {
        if method != "POST" && method != "PUT" {
            body := "method not allowed\n"
            write_http(fd, 405, "text/plain; charset=utf-8", body) or (err) { << }
            net.conn_close(fd) or (err) { << }
            <<
        }
        echo := req.body()
        write_http(fd, 200, "text/plain; charset=utf-8", echo) or (err) { << }
        net.conn_close(fd) or (err) { << }
        <<
    }
    miss := "not found: " + path + "\n"
    write_http(fd, 404, "text/plain; charset=utf-8", miss) or (err) { << }
    net.conn_close(fd) or (err) { << }
}

fn main() int {
    mut s := http.new_server()
    s.listen("127.0.0.1:18080") or (err) {
        println(err.msg())
        << 1
    }
    println("http://127.0.0.1:18080")
    mut running := true
    while running {
        fd := s.accept() or (err) {
            ->
        }
        // The deployable example uses the portable, deterministic path on all
        // hosts. The stackful `xu` scheduler has dedicated stress fixtures,
        // but it is not yet a production request-execution primitive; keeping
        // request handling sequential also gives local and server builds the
        // same lifecycle contract. Put TLS, limits and concurrency in the
        // reverse proxy until that runtime gate is closed.
        handle(fd)
    }
    << 0
}
