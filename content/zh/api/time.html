fn (m $&Model) work(screen $&widgets.Screen, now time.Instant) int {
    if !m.running || now.before(m.next_work) { << 0}
    // Each progress update follows an actual bounded work batch. The pause
    // represents an application work queue and never changes the percentage.
    for i in 0:5000 {m.checksum =(m.checksum + i *(m.completed + 1)) % 1000003}
    m.completed = m.completed + 1
    _ := screen.set_task_progress(m.task, m.task_id, f64(m.completed) / 48.0) or >>
    m.next_work = now.add(time.milliseconds(35) or >>) or >>
    if m.fail_next && m.completed == 18 {
        _ := screen.complete_task(m.task, m.task_id, widgets.TASK_ERROR) or >>
        m.running = false
        m.fail_next = false
        screen.set_text(m.message, "Example failure. Retry starts a new task; stale progress is rejected.") or >>
    } else if m.completed == 48 {
        _ := screen.complete_task(m.task, m.task_id, widgets.TASK_SUCCESS) or >>
        m.running = false
        screen.set_text(m.message, "48/48 work batches completed · checksum " + str(m.checksum)) or >>
    }
    << 0
}

    $selected := 0.0

fn (m &Model) next_frame(now time.Instant) time.Instant{
    if m.switch_motion.is_animating() || m.popup_motion.is_animating() { << now.add(time.milliseconds(16) or >>) or >> }
    for motion in m.transitions{if motion.is_animating() { << now.add(time.milliseconds(16) or >>) or >> }}
    $deadline := time.zero_instant()
    for next in[m.input.next_caret_tick(now, m.reduced_motion) or >>, m.invalid.next_caret_tick(now, m.reduced_motion) or >>,
    m.read_only.next_caret_tick(now, m.reduced_motion) or >>, m.area.next_caret_tick(now, m.reduced_motion) or >>] {
        if !next.is_zero() &&(deadline.is_zero() || next.before(deadline)) {deadline = next}
    }
    << deadline
}
fn (m &Model) next_event(ech chan ui.Event) ui.Event{
    deadline := m.next_frame(time.instant()) or >>
    while true {
        event := ui.poll_chan(ech)
        if event.kind != ui.EV_NONE{ << event}
        if !deadline.is_zero() && !time.instant().before(deadline) { << event}
        pause := time.milliseconds(8) or >>
        pause.sleep()
    }
    << ui.Event{kind: ui.EV_NONE, window_id: 0, menu_id: "", width: 0, height: 0}
}

fn (m $&Model) dispatch_input(view $&widgets.View, event &ui.InputEvent)[]widgets.Action{
    was_open := m.select_ctrl.is_open()
    pointer := event.kind == ui.EV_POINTER_DOWN || event.kind == ui.EV_POINTER_UP ||
    event.kind == ui.EV_POINTER_MOVE || event.kind == ui.EV_POINTER_SCROLL
    if pointer &&(was_open || m.select_popup_armed) {
        local := view.local_input(41, event) or >>
        styles := view.styles_for(41) or >>
        pop := m.select_ctrl.popup_bounds(&styles)
        if pop.contains(local.x, local.y) || m.select_popup_armed{
            if event.kind == ui.EV_POINTER_DOWN && event.button == ui.BUTTON_LEFT{m.select_popup_armed = true}
            control := $&m.select_ctrl
            action := control.input(&local) or >>
            view.set_pointer_cursor(ui.CURSOR_HAND) or >>
            if event.kind == ui.EV_POINTER_UP{m.select_popup_armed = false}
            view.mark_paint(41) or >>
            view.mark_paint(1) or >>
            <<[action]
        }
    }
    if event.kind == ui.EV_POINTER_CANCEL{m.select_popup_armed = false}
    actions := widgets.dispatch(view, event, m) or >>
    if was_open || m.select_ctrl.is_open() {view.mark_paint(1) or >> }
    << actions
}

fn chart_data(kind int, revision int)[]widgets.ChartSeries{
    $result :=[]widgets.ChartSeries{}
    $series_count := 2
    if kind == widgets.CHART_DONUT{series_count = 1}
    for si in 0:series_count{
        $name := "Build 构建"
        if si == 1 {name = "Index 索引"}
        if kind == widgets.CHART_DONUT{name = "Workload 工作量"}
        $series := widgets.chart_series(si + 1, name, kind)
        series.smooth = kind == widgets.CHART_LINE || kind == widgets.CHART_AREA
        $count := 12 - revision % 2
        if kind == widgets.CHART_SCATTER{count = 64}
        if kind == widgets.CHART_DONUT{count = 5}
        for i in 0:count{
            x := f64(i + 1)
            phase := f64(revision) * 0.9 + f64(si) * 1.7
            y := 48.0 + f64(si) * 20.0 + math.sin(x * 0.55 + phase) * 20.0 + math.cos(x * 0.17 + phase) * 7.0
            $point := widgets.chart_point(i + 1, x, y)
            point.label = "Sample 样本 " + str(i + 1)
            if kind == widgets.CHART_DONUT{point.label =["Compile 编译", "Index 索引", "Render 绘制", "I/O 输入输出", "Other 其他"][i]}
            series.points.push(point)
        }
        result.push(series)
    }
    << result
}
