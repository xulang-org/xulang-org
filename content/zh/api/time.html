fn (m &mut Model) motion_style(id int,base &widgets.ControlStyle,state int) widgets.ControlStyle {
    now := m.clock
    reduced := m.reduced_motion

    << motion.style(base,state,now,reduced) or >>
}
fn (m &mut Model) tick(view &mut widgets.View,now time.Instant) int {
    m.clock = now
    reduced := m.reduced_motion
    for id in 1:m.transitions.len {
        motion := mut &m.transitions[id]
        if reduced && motion.is_animating() { motion.finish()
            view.mark_paint(id) or >> }
        else if motion.advance(now) or >> { view.mark_paint(id) or >> }
    }
    mut selected := 0.0
    if m.toggle.value() { selected = 1.0 }
    switch_motion := mut &m.switch_motion
    before_switch := switch_motion.value()
    switch_motion.retarget(selected,now,time.milliseconds(180) or >>,reduced) or >>
    if before_switch != switch_motion.value() { view.mark_paint(32) or >> }
    mut opened := 0.0
    if m.select_ctrl.is_open() { opened = 1.0 }
    popup_motion := mut &m.popup_motion
    before_popup := popup_motion.value()
    mut popup_duration := time.milliseconds(130) or >>
    if opened == 0.0 { popup_duration = time.milliseconds(95) or >> }
    popup_motion.retarget(opened,now,popup_duration,reduced) or >>
    if before_popup != popup_motion.value() { view.mark_paint(1) or >> }
    input := mut &m.input
    if input.tick(now,reduced) or >> { view.mark_paint(12) or >> }
    invalid := mut &m.invalid
    if invalid.tick(now,reduced) or >> { view.mark_paint(13) or >> }
    read_only := mut &m.read_only
    if read_only.tick(now,reduced) or >> { view.mark_paint(14) or >> }
    area := mut &m.area
    if area.tick(now,reduced) or >> { view.mark_paint(15) or >> }
    << 0
}
fn (m &Model) next_frame(now time.Instant) time.Instant {
    if m.switch_motion.is_animating() || m.popup_motion.is_animating() { << now.add(time.milliseconds(16) or >>) or >> }
    for motion in m.transitions { if motion.is_animating() { << now.add(time.milliseconds(16) or >>) or >> } }
    mut deadline := time.zero_instant()
    for next in [m.input.next_caret_tick(now,m.reduced_motion) or >>,m.invalid.next_caret_tick(now,m.reduced_motion) or >>,
        m.read_only.next_caret_tick(now,m.reduced_motion) or >>,m.area.next_caret_tick(now,m.reduced_motion) or >>] {
        if !next.is_zero() && (deadline.is_zero() || next.before(deadline)) { deadline = next }
    }
    << deadline
}
fn (m &Model) next_event(ech chan ui.Event) ui.Event {
    deadline := m.next_frame(time.instant()) or >>
    while true {
        event := ui.poll_chan(ech)
        if event.kind != ui.EV_NONE { << event }
        if !deadline.is_zero() && !time.instant().before(deadline) { << event }
        pause := time.milliseconds(8) or >>
        pause.sleep()
    }
    << ui.Event{kind: ui.EV_NONE,window_id: 0,menu_id: "",width: 0,height: 0}
}

fn (m &mut Model) paint_select_popup(s &widgets.Styles,f &mut draw.Frame) int {
    mut control := m.select_ctrl
    control.set_open(true)
    opacity := m.popup_motion.value()
    if opacity <= 0.0 { << 0 }
    base := control.style_or(s)
    appearance := base.with_opacity(opacity) or >>
    control.set_style(&appearance) or >>
    popup := control.popup_bounds(s)
    mut offset := -4.0 * (1.0 - opacity)
    if popup.y < 0.0 { offset = 4.0 * (1.0 - opacity) }
    f.push() or >>
    f.translate(m.select_origin.x,m.select_origin.y + offset) or >>
    control.paint_popup_box_styled(f,s) or >>
    auto_label := mut &m.option_auto
    control.paint_option_styled(f,0,auto_label,s) or >>
    manual_label := mut &m.option_manual
    control.paint_option_styled(f,1,manual_label,s) or >>
    locked_label := mut &m.option_locked
    control.paint_option_styled(f,2,locked_label,s) or >>
    f.pop() or >>
    << 0
}

fn (m &mut Model) dispatch_input(view &mut widgets.View,event &ui.InputEvent) []widgets.Action {
    was_open := m.select_ctrl.is_open()
    pointer := event.kind == ui.EV_POINTER_DOWN || event.kind == ui.EV_POINTER_UP ||
        event.kind == ui.EV_POINTER_MOVE || event.kind == ui.EV_POINTER_SCROLL
    if pointer && (was_open || m.select_popup_armed) {
        local := view.local_input(41,event) or >>
        styles := view.styles_for(41) or >>
        pop := m.select_ctrl.popup_bounds(&styles)
        if pop.contains(local.x,local.y) || m.select_popup_armed {
            if event.kind == ui.EV_POINTER_DOWN && event.button == ui.BUTTON_LEFT { m.select_popup_armed = true }
            control := mut &m.select_ctrl
            action := control.input(&local) or >>
            view.set_pointer_cursor(ui.CURSOR_HAND) or >>
            if event.kind == ui.EV_POINTER_UP { m.select_popup_armed = false }
            view.mark_paint(41) or >>
            view.mark_paint(1) or >>
            << [action]
        }
    }
    if event.kind == ui.EV_POINTER_CANCEL { m.select_popup_armed = false }
    actions := widgets.dispatch(view,event,m) or >>
    if was_open || m.select_ctrl.is_open() { view.mark_paint(1) or >> }
    << actions
}
fn (m &mut Model) dispatch_event(view &mut widgets.View,event &ui.Event) []widgets.Action {
    if ui.is_input_event(event.kind) {
        input := ui.input_of(event) or >>
        << m.dispatch_input(view,&input) or >>
    }
    was_open := m.select_ctrl.is_open()
    actions := widgets.dispatch_event(view,event,m) or >>
    m.select_popup_armed = false
    if was_open && !m.select_ctrl.is_open() { view.mark_paint(1) or >> }
    << actions
}

fn (m &mut Model) place(view &mut widgets.View) int {
    select_bounds := view.node_bounds(41) or >>
    m.select_origin = select_bounds
    select_styles := view.styles_for(41) or >>
    select_control := mut &m.select_ctrl
    select_control.set_bounds(&layout.Rect{x: 0.0,y: 0.0,width: select_bounds.width,height: select_bounds.height}) or >>
    select_control.apply_styles(&select_styles) or >>
    popup := select_control.popup_bounds(&select_styles)
    window_bounds := view.node_bounds(1) or >>
    below := window_bounds.y + window_bounds.height - select_bounds.y - select_bounds.height
    above := select_bounds.y - window_bounds.y
    select_control.set_popup_above(below < popup.height + 2.0 * select_styles.density && above > below)
    list_bounds := view.node_bounds(43) or >>

fn (m &mut Model) handle_action(view &mut widgets.View,action &widgets.Action) int {
    if action.kind == widgets.ACTION_ACTIVATE {
        if action.id == 4 {
            switcher := mut &m.switcher
            _ := switcher.cycle(view) or >>
        }
        if action.id == 5 {
            switcher := mut &m.switcher
            mut next_scale := 1.4
            if switcher.text_scale() != 1.0 { next_scale = 1.0 }
            switcher.set_text_scale(next_scale) or >>
            switcher.apply(view) or >>
        }
        if action.id == 6 {
            switcher := mut &m.switcher
            mut next_density := 0.8
            if switcher.density() != 1.0 { next_density = 1.0 }
            switcher.set_density(next_density) or >>
            switcher.apply(view) or >>
        }
        if action.id == 41 {
            sel := mut &m.select_ctrl
            next_opt := (sel.selected() + 1) % 3
            sel.set_selected(next_opt) or >>
            lbl := mut &m.select_label
            if next_opt == 0 { lbl.set_text(&"Option: Auto") }
            if next_opt == 1 { lbl.set_text(&"Option: Manual") }
            if next_opt == 2 { lbl.set_text(&"Option: Locked") }
            view.mark_measure(41) or >>
            view.mark_paint(41) or >>
        }
    }
    if action.id == 16 && action.kind == widgets.ACTION_SCROLL { view.set_scroll(16,0.0,action.value) or >> }
    if action.kind == widgets.ACTION_CHANGE {
        if action.id == 32 {
            bdg := mut &m.badge
            bdg.set_selected(action.value > 0.5)
            blbl := mut &m.badge_label
            if action.value > 0.5 { blbl.set_text(&"Active") } else { blbl.set_text(&"Paused") }
            view.mark_measure(37) or >>
            view.mark_paint(37) or >>
        }
        if action.id == 33 {
            rb := mut &m.radio_b
            rb.set_value(false)
            view.mark_paint(34) or >>
        }
        if action.id == 34 {
            ra := mut &m.radio_a
            ra.set_value(false)
            view.mark_paint(33) or >>
        }
        if action.id == 35 {
            prog := mut &m.progress
            prog.set_value(action.value / 100.0) or >>
            view.mark_paint(11) or >>
        }
        if action.id == 36 {
            lbl := mut &m.stepper_label
            lbl.set_text(&str(int(action.value)))
            view.mark_measure(36) or >>
            prog := mut &m.progress
            prog.set_value(action.value / 100.0) or >>
            view.mark_paint(11) or >>
        }
        if action.id == 40 {
            clbl := mut &m.card_label
            if action.value == 0.0 { clbl.set_text(&"Card · Overview") }
            if action.value == 1.0 { clbl.set_text(&"Card · Tokens") }
            if action.value == 2.0 { clbl.set_text(&"Card · Segmented") }
            view.mark_measure(42) or >>
            view.mark_paint(42) or >>
        }
        if action.id == 41 {
            lbl := mut &m.select_label
            if action.value == 0.0 { lbl.set_text(&"Option: Auto") }
            if action.value == 1.0 { lbl.set_text(&"Option: Manual") }
            if action.value == 2.0 { lbl.set_text(&"Option: Locked") }
            view.mark_measure(41) or >>
            view.mark_paint(41) or >>
        }
        if action.id == 43 {
            clbl := mut &m.card_label
            clbl.set_text(&("Card · Row " + str(int(action.value) + 1)))
            view.mark_measure(42) or >>
            view.mark_paint(42) or >>
        }
    }
    << 0
}
