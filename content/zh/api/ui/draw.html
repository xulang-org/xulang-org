fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.ev_exit { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    mut frame := context.new_frame(4096)
    mut hits := 0
    mut pressed := false
    paint(&context,mut &frame,hits) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.ev_exit { <- }
        if event.window_id != identity.window_id { -> }
        if event.kind == ui.ev_canvas_closed { <- }
        if event.kind == ui.ev_blurred { pressed = false }
        if event.kind == ui.ev_canvas_metrics { paint(&context,mut &frame,hits) or >> }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            if input.kind == ui.ev_key_down && input.key == ui.key_escape { demo.quit() or >> }
            dx := input.x - 112.0
            dy := input.y - 228.0
            inside := dx * dx + dy * dy <= 2704.0
            if input.kind == ui.ev_pointer_down && input.button == ui.button_left { pressed = inside }
            if input.kind == ui.ev_pointer_cancel { pressed = false }
            if input.kind == ui.ev_pointer_up && input.button == ui.button_left {
                if pressed && inside { hits = hits + 1
                    paint(&context,mut &frame,hits) or >> }
                pressed = false
            }
        }
    }
    frame.dispose()
    << 0
}

fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.ev_exit { << 0 }
    identity := ui.canvas_of(&ready) or >>
    mut app := demo.windows(ech)
    first := ui.Window{id: identity.window_id,generation: identity.generation,title: "Xu UI / Window 1",width: 560,height: 340,content_kind: ui.content_draw,content_backend_id: identity.canvas_id,closed: 0}
    mut windows := []ui.Window{}
    windows.push(first)
    mut counts := [0]
    paint(windows[0],0) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.ev_exit { <- }
        mut index := -1
        for i in 0:windows.len { if windows[i].id == event.window_id { index = i } }
        if index < 0 { -> }
        if event.kind == ui.ev_canvas_ready {
            window := windows[index]
            window.content_command("input_enabled\t1") or >>
            window.content_command("pan_enabled\t0") or >>
            println("Canvas ready: " + str(event.window_id))
            paint(window,counts[index]) or >> }
        if event.kind == ui.ev_canvas_metrics { paint(windows[index],counts[index]) or >> }
        if event.kind == ui.ev_closed { windows[index].closed = 1 }
        if event.kind == ui.ev_window_failed { windows[index].closed = 1
            eprintln("Native window creation failed: " + event.menu_id) }
        if !ui.is_input_event(event.kind) { -> }
        input := ui.input_of(&event) or >>
        if input.generation != windows[index].generation || windows[index].closed != 0 { -> }
        if input.kind == ui.ev_pointer_up && input.button == ui.button_left {
            counts[index] = counts[index] + 1
            println("Window " + str(event.window_id) + " count: " + str(counts[index]))
            paint(windows[index],counts[index]) or >>
        }
        if input.kind != ui.ev_key_down || input.repeat != 0 { -> }
        if input.key == ui.key_escape { demo.quit() or >> }
        if input.key == 78 || input.key == 110 {
            // At most six demo windows; the host's own limit is independent.
            mut live := 0
            for window in windows { if window.closed == 0 { live = live + 1 } }
            if live < 6 {
                window := app.open_content_window(ui.WindowOpts{title: "Xu UI / Additional Window",width: 560,height: 340},ui.content_draw) or >>
                windows.push(window)
                counts.push(0)
            }
        }
        if input.key == 87 || input.key == 119 {
            mut window := windows[index]
            window.request_close() or >>
            windows[index] = window
        }
        if input.key == 82 || input.key == 114 {
            mut window := windows[index]
            window.resize(660,400) or >>
            windows[index] = window
        }
        if input.key == 70 || input.key == 102 {
            for window in windows { if window.closed == 0 { window.focus() or >>
                <- } }
        }
    }
    << 0
}

fn record(context &draw.Context,image &draw.Image) draw.DisplayList {
    mut fragment := context.new_frame(4096)
    fragment.begin() or >>
    fragment.fill_rect(0.0,0.0,960.0,510.0,draw.rgb(26,37,50)) or >>
    demo.text(context,mut &fragment,24.0,22.0,18.0,demo.ink(),"Vectors / gradients / embedded PNG / shaped text") or >>
    mut path := draw.new_path(256)
    path.move_to(40.0,210.0) or >>
    path.cubic_to(80.0,60.0,210.0,60.0,250.0,210.0) or >>
    path.quad_to(145.0,300.0,40.0,210.0) or >>
    path.close() or >>
    stops := [draw.GradientStop{offset: 0.0,color: demo.accent()},draw.GradientStop{offset: 1.0,color: draw.rgb(92,127,203)}]
    gradient := draw.linear_gradient(20.0,80.0,260.0,270.0,stops)
    bounds := draw.ImageRect{x: 20.0,y: 70.0,width: 260.0,height: 230.0}
    fragment.fill_path_gradient(&path,&gradient,&bounds,2.0,draw.fill_nonzero) or >>
    stroke := draw.stroke_options(2.0)
    fragment.stroke_path(&path,demo.ink(),&stroke) or >>
    fragment.image(image,320.0,88.0,150.0,150.0) or >>
    source := draw.ImageRect{x: 0.0,y: 0.0,width: f64(image.width),height: f64(image.height)}
    destination := draw.ImageRect{x: 510.0,y: 88.0,width: 190.0,height: 150.0}
    fragment.image_region(image,&source,&destination,0.45,draw.sampler_nearest) or >>

    fragment.text_layout(&text,32.0,330.0,demo.ink()) or >>
    println("Text metrics: " + str(text.metrics.lines) + " lines")
    layer := draw.ImageRect{x: 740.0,y: 80.0,width: 180.0,height: 200.0}

fn paint(context &draw.Context,frame &mut draw.Frame,list &draw.DisplayList,cache &mut draw.SurfaceCache,density f64,cached bool) int {
    frame.begin() or >>
    frame.set_bg(demo.background()) or >>
    demo.text(context,frame,32.0,24.0,14.0,demo.accent(),"02 / RESOURCE GALLERY") or >>
    demo.text(context,frame,32.0,54.0,28.0,demo.ink(),"Record once. Reuse across frames.") or >>
    frame.push() or >>
    frame.translate(32.0,110.0) or >>
    if cached { frame.cached_display_list(cache,list,0.0,0.0,960.0,510.0,density,1.0,draw.sampler_linear) or >> }
    else { frame.display_list(list) or >> }
    frame.pop() or >>
    mut mode := "vector replay"
    if cached { mode = "raster cache" }
    demo.text(context,frame,32.0,646.0,15.0,demo.ink(),"Space: vector/cache / R: invalidate / current: " + mode) or >>
    frame.present() or >>
    println("Presented " + mode + ": " + str(frame.command_count()) + " frame commands")
    << 0
}
fn scene(ech chan ui.Event) int {
    ready := demo.ready(ech) or >>
    if ready.kind == ui.ev_exit { << 0 }
    identity := ui.canvas_of(&ready) or >>
    context := draw.context_of(&ready) or >>
    path := "sample.png"
    mut image := context.load_embedded_image(&path) or >>
    mut list := record(&context,&image) or >>
    mut cache := draw.new_surface_cache()
    mut frame := context.new_frame(1024)
    mut density := 1.0
    mut cached := false
    paint(&context,mut &frame,&list,mut &cache,density,cached) or >>
    while true {
        event := demo.next(ech)
        if event.kind == ui.ev_exit || event.kind == ui.ev_canvas_closed { <- }
        if event.window_id != identity.window_id { -> }
        mut dirty := false
        if event.kind == ui.ev_canvas_metrics {
            metrics := ui.metrics_of(&event) or >>
            density = metrics.scale_x
            dirty = true
        }
        if ui.is_input_event(event.kind) {
            input := ui.input_of(&event) or >>
            if input.generation != identity.generation { -> }
            if input.kind == ui.ev_key_down && input.repeat == 0 {
                if input.key == 32 { cached = !cached
                    dirty = true }
                if input.key == 82 || input.key == 114 { cache.invalidate()
                    dirty = true }
                if input.key == ui.key_escape { demo.quit() or >> }
            }
        }
        if dirty { paint(&context,mut &frame,&list,mut &cache,density,cached) or >> }
    }
    frame.dispose()
    cache.dispose()
    // Native close has already revoked all canvas-owned resources.
    if demo.alive(&identity) {
        list.dispose() or >>
        image.dispose() or >>
    }
    << 0
}
