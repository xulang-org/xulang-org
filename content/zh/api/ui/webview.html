fn route(request webview.InvokeRequest) string {
    if request.name == "slow" { sleep_ns(300000000) }
    if request.name == "wait" { sleep_ns(1500000000) }
    if request.name == "native" {
        target := ui.CanvasEvent{window_id: request.window_id,canvas_id: request.window_id,generation: request.generation}
        options := ui.RequestOptions{timeout_ms: 2000}
        script := "document.title"
        result := ui.webview_request_wait(&target,ui.REQUEST_WEBVIEW_EVAL,&script,&options) or (err) { << "failed: " + err.msg() }
        if result.status != ui.REQUEST_COMPLETED { << "status: " + str(result.status) }
        << result.value + ""
    }
    << request.payload + ""
}
fn ipc(emit_ch chan string,ech chan ui.Event) {
    webview.bind_ipc_requests_www(":mem:",route,emit_ch,ech) or (err) { eprintln(err.msg())
        << }
}

    while true { event := <-ech
        if event.kind == ui.EV_EXIT { emit_ch <- ""
            << }
        if event.kind == ui.EV_CLOSED { println("Window closed; pending document requests revoked") }
    }
}

fn tick_loop(emit_ch chan string) {
    $n := 0
    while true {
        sleep_ns(3000000000)
        n = n + 1
        webview.bus_emit(emit_ch, "tick", str(n))
    }
}

fn watch_shell(shell_ch chan ui.Event) {
    while true {
        ev := <-shell_ch
        if ev.kind == ui.EV_EXIT || ev.kind == ui.EV_CLOSED {
            println("shell event kind=" + str(ev.kind))
        } else if ev.kind == ui.EV_LOADED {
            println("shell event loaded")
        } else if ev.kind == ui.EV_LOAD_FAILED {
            println("shell event load_failed")
        } else if ev.kind == ui.EV_READY {
            println("shell event ready")
        }
    }
}

fn main() int {
    emit_ch := chan string{cap: 32}
    shell_ch := chan ui.Event{cap: 32}
    $app := ui.new_app_events(shell_ch)
    app.set_icon(ui.icon_from_file("brand/xu_app_logo.png"))
    url := remote_url()
    xu watch_shell(shell_ch)
    xu ipc_boot(emit_ch, shell_ch)
    xu tick_loop(emit_ch)
    sleep_ns(50000000)
    $win := app.new_window(ui.WindowOpts{title: "Xu Remote HTTP + xu:// IPC", width: 720, height: 520})

fn next_event(events chan ui.Event, deadline time.Instant) ui.Event{
    if deadline.is_zero() { << <- events}
    delay := deadline.since(time.instant())
    if delay.nanoseconds() <= 0 { << ui.Event{kind: ui.EV_NONE, window_id: 0, menu_id: "", width: 0, height: 0}}
    timer := delay.timer()
    select {event := <- events{timer.stop()
            << event}
        _ := <- timer.c{timer.stop()}}
    << ui.Event{kind: ui.EV_NONE, window_id: 0, menu_id: "", width: 0, height: 0}
}
