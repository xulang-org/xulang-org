fn verify_postgres_pool() {
    println("\n--- Testing Native Postgres Connection Pool ---")
    dsn := "postgres://zicq:zicq@127.0.0.1:5432/xu_demo?sslmode=disable"
    $pool := postgres.open_pool(dsn + "", 5, 2)

    pool.ping() or (err) {
        println("   [FAIL] Pool ping error: " + err.msg())
        <<
    }
    println("   [PASS] Postgres Pool ping succeeded.")

    // 1. DDL
    pool.exec("DROP TABLE IF EXISTS xu_pool_users;") or (err) {
        println("   [FAIL] Drop table error: " + err.msg())
        <<
    }
    pool.exec("CREATE TABLE xu_pool_users (id INTEGER PRIMARY KEY, name VARCHAR(50), balance INTEGER);") or (err) {
        println("   [FAIL] Create table error: " + err.msg())
        <<
    }
    println("   [PASS] Pool created table xu_pool_users.")

    // 2. Concurrency / multiple inserts
    $i := 1
    while i <= 5 {
        pool.exec("INSERT INTO xu_pool_users (id, name, balance) VALUES ($$1, $$2, $$3);", i, "User_" + str(i), i * 100) or (err) {
            println("   [FAIL] Insert row error: " + err.msg())
            <<
        }
        i = i + 1
    }
    println("   [PASS] Concurrently inserted 5 rows across pooled connections.")

    // 3. Query
    $rows := pool.query("SELECT id, name, balance FROM xu_pool_users ORDER BY id;") or (err) {
        println("   [FAIL] Pool query error: " + err.msg())
        <<
    }
    $count := 0
    while rows.next() {
        vals := rows.take() or (err) {
            println("   [FAIL] rows.take error: " + err.msg())
            <<
        }
        uid := vals[0] as int
        uname := vals[1] as string
        ubal := vals[2] as int
        println("      Row[" + str(count) + "]: id=" + str(uid) + ", name=" + uname + ", bal=" + str(ubal))
        count = count + 1
    }
    rows.close()
    if count == 5 {
        println("   [PASS] Pool query verified 5 rows.")
    } else {
        println("   [FAIL] Row count mismatch: " + str(count))
    }

    // 4. Transaction via Pool
    $tx := pool.begin() or (err) {
        println("   [FAIL] Pool begin tx error: " + err.msg())
        <<
    }
    tx.exec("INSERT INTO xu_pool_users (id, name, balance) VALUES ($$1, $$2, $$3);", 6, "User_6", 600) or (err) {
        println("   [FAIL] Tx insert error: " + err.msg())
        <<
    }
    tx.commit() or (err) {
        println("   [FAIL] Tx commit error: " + err.msg())
        <<
    }
    println("   [PASS] Transaction committed via pool.")

    // 5. Cleanup & Close
    pool.exec("DROP TABLE IF EXISTS xu_pool_users;") or (err) {
        println("   [FAIL] Cleanup table error: " + err.msg())
        <<
    }
    pool.close() or (err) {
        println("   [FAIL] Pool close error: " + err.msg())
        <<
    }
    println("   [PASS] Postgres Pool closed cleanly.")
}
